def compute_rsi(series, period=14):
    delta    = series.diff()
    gain     = delta.clip(lower=0)
    loss     = -delta.clip(upper=0)
    avg_gain = gain.ewm(com=period-1, min_periods=period).mean()
    avg_loss = loss.ewm(com=period-1, min_periods=period).mean()
    rs       = avg_gain / avg_loss.replace(0, np.nan)
    return 100 - (100 / (1 + rs))

def compute_r2(series, length):
    return series.rolling(length).apply(
        lambda x: np.corrcoef(x, np.arange(len(x)))[0,1]**2
        if not np.isnan(x).any() else np.nan, raw=True)

def build_indicators(df):
    df = df.copy()
    df["rsi"]     = compute_rsi(df["Close"], 14)
    df["sma200"]  = df["Close"].rolling(200).mean()
    df["sma50"]   = df["Close"].rolling(50).mean()
    df["vol_sma"] = df["Volume"].rolling(20).mean()
    df["r2"]      = compute_r2(df["Close"], CONFIG["r2_len"])
    return df

# 3. BACKTEST ENGINE
# ══════════════════════════════════════════════════════════════════

def run_backtest(df, cfg):
    equity         = float(cfg["initial_capital"])
    leverage       = float(cfg["leverage"])
    pos_size       = 0.0
    avg_entry      = 0.0
    open_trades    = 0
    last_loss_bar  = -999
    daily_start_eq = equity
    profit_taken   = False
    current_day    = None
    trail_high     = 0.0
    entry_date     = None   # date of the T1 entry bar
    trades         = []
    equity_curve   = []
    stop_curve     = []

    closes  = df["Close"].values
    opens   = df["Open"].values   if "Open"   in df.columns else closes
    highs   = df["High"].values   if "High"   in df.columns else closes
    lows    = df["Low"].values    if "Low"    in df.columns else closes
    volumes = df["Volume"].values
    sma200  = df["sma200"].values
    sma50   = df["sma50"].values
    vol_sma = df["vol_sma"].values
    r2_arr  = df["r2"].values
    rsi_arr = df["rsi"].values
    dates   = df.index  # real datetime index

    xu40 = np.zeros(len(df), dtype=bool)
    xu32 = np.zeros(len(df), dtype=bool)
    xu25 = np.zeros(len(df), dtype=bool)
    xo50 = np.zeros(len(df), dtype=bool)
    for i in range(1, len(df)):
        if not (np.isnan(rsi_arr[i]) or np.isnan(rsi_arr[i-1])):
            xu40[i] = rsi_arr[i-1] >= cfg["t1_rsi"] and rsi_arr[i] < cfg["t1_rsi"]
            xu32[i] = rsi_arr[i-1] >= cfg["t2_rsi"] and rsi_arr[i] < cfg["t2_rsi"]
            xu25[i] = rsi_arr[i-1] >= cfg["t3_rsi"] and rsi_arr[i] < cfg["t3_rsi"]
            xo50[i] = rsi_arr[i-1] <= cfg["exit_rsi"] and rsi_arr[i] > cfg["exit_rsi"]

    def get_stop_pct():
        if open_trades == 3: return cfg.get("stop_pct_t3", 3.0) / 100
        if open_trades == 2: return cfg.get("stop_pct_t2", 4.0) / 100
        return cfg.get("stop_pct_t1", 5.0) / 100

    def calc_stop(avg_px, cur_px, t_hi):
        base   = avg_px * (1 - get_stop_pct())
        be_pct = cfg.get("breakeven_pct", 2.0) / 100
        profit_pct = (cur_px - avg_px) / avg_px if avg_px > 0 else 0

        # Move stop to break-even once price rises be_pct above entry
        if profit_pct > be_pct:
            base = max(base, avg_px)

        # Only apply trailing stop once we are above break-even
        # This prevents trail from firing immediately at entry
        if profit_pct > be_pct:
            trail = t_hi * (1 - cfg["trail_perc"] / 100)
            return max(base, trail)

        return base

    for i in range(len(df)):
        close  = float(closes[i])
        volume = float(volumes[i])
        day    = dates[i].date() if hasattr(dates[i], "date") else dates[i]

        if day != current_day:
            daily_start_eq = equity
            profit_taken   = False
            current_day    = day

        is_cool       = (i - last_loss_bar) < cfg["cooldown_bars"]
        daily_ok      = (daily_start_eq - equity) <= daily_start_eq * cfg["max_daily_loss"] / 100
        above_200     = bool(close > sma200[i]) if not np.isnan(sma200[i]) else False
        is_ranging    = bool(r2_arr[i] < cfg["max_r2"]) if not np.isnan(r2_arr[i]) else False

        # SMA 200 slope filter — only enter when SMA 200 is flat or rising
        slope_on   = str(cfg.get("sma200_slope_on", "true")).lower() not in ("false","0","no","off")
        slope_bars = int(cfg.get("sma200_slope_bars", 10))
        slope_min  = float(cfg.get("sma200_slope_min", 0.0))
        if slope_on and i >= slope_bars and not np.isnan(sma200[i]) and not np.isnan(sma200[i - slope_bars]):
            sma200_slope_pct = (sma200[i] - sma200[i - slope_bars]) / sma200[i - slope_bars] * 100
            slope_ok = sma200_slope_pct >= slope_min
        else:
            slope_ok = True

        # SMA 50 slope filter — only enter when SMA 50 is flat or rising
        s50_on   = str(cfg.get("sma50_slope_on", "true")).lower() not in ("false","0","no","off")
        s50_bars = int(cfg.get("sma50_slope_bars", 5))
        s50_min  = float(cfg.get("sma50_slope_min", 0.0))
        if s50_on and i >= s50_bars and not np.isnan(sma50[i]) and not np.isnan(sma50[i - s50_bars]):
            sma50_slope_pct = (sma50[i] - sma50[i - s50_bars]) / sma50[i - s50_bars] * 100
            s50_slope_ok = sma50_slope_pct >= s50_min
        else:
            s50_slope_ok = True

        can_enter = above_200 and not is_cool and daily_ok and is_ranging and slope_ok and s50_slope_ok
        is_hi_vol     = bool(volume > vol_sma[i] * cfg["vol_mult"]) if not np.isnan(vol_sma[i]) else False

        if pos_size > 0:
            trail_high   = max(trail_high, close)
            current_stop = calc_stop(avg_entry, close, trail_high)
            if close <= current_stop:
                pnl = (close - avg_entry) * pos_size
                equity += pnl
                trades.append({"type":"STOP","date":dates[i],"entry_date":entry_date,
                                "entry":avg_entry,"exit":close,
                                "qty":pos_size,"pnl":pnl,"bar":i})
                if pnl < 0: last_loss_bar = i
                pos_size = avg_entry = trail_high = 0.0
                open_trades = 0; profit_taken = False
                entry_date  = None
                equity_curve.append(equity); stop_curve.append(np.nan)
                continue

        if can_enter:
            if xu40[i] and pos_size == 0:
                qty = (equity * cfg.get("t1_size", 1.0) * leverage) / close
                pos_size += qty; avg_entry = close
                open_trades = 1; trail_high = close
                entry_date = dates[i]   # record when this trade opened
                trades.append({"type":"T1_ENTRY","date":dates[i],"price":close,"bar":i})
            elif pos_size > 0 and xu32[i] and open_trades == 1:
                qty = (equity * cfg.get("t2_size", 2.0) * leverage) / close
                avg_entry = (pos_size*avg_entry + qty*close) / (pos_size+qty)
                pos_size += qty; open_trades = 2
                trades.append({"type":"T2","date":dates[i],"price":close,"bar":i})
            elif pos_size > 0 and xu25[i] and open_trades == 2:
                qty = (equity * cfg.get("t3_size", 2.0) * leverage) / close
                avg_entry = (pos_size*avg_entry + qty*close) / (pos_size+qty)
                pos_size += qty; open_trades = 3
                trades.append({"type":"T3","date":dates[i],"price":close,"bar":i})

        if pos_size > 0:
            ab50      = bool(close > sma50[i]) if not np.isnan(sma50[i]) else False
            exit_trig = bool(xo50[i]) or ab50

            if exit_trig and not profit_taken:
                # Always exit 100% of position
                pnl      = (close - avg_entry) * pos_size
                equity  += pnl
                trades.append({"type":"EXIT","date":dates[i],"entry_date":entry_date,
                                "entry":avg_entry,"exit":close,
                                "qty":pos_size,"pnl":pnl,"bar":i})
                pos_size     = 0.0
                avg_entry    = 0.0
                trail_high   = 0.0
                open_trades  = 0
                profit_taken = False
                entry_date   = None

        mtm = equity + (pos_size * (close - avg_entry) if pos_size > 0 else 0)
        equity_curve.append(mtm)
        stop_curve.append(calc_stop(avg_entry, close, max(trail_high, close))
                          if pos_size > 0 else np.nan)

    return equity_curve, stop_curve, trades


def compute_metrics(equity_curve, trades, initial_capital):
    eq           = np.array(equity_curve)
    total_return = (eq[-1] - initial_capital) / initial_capital * 100
    peak         = np.maximum.accumulate(eq)
    drawdown     = (eq - peak) / peak * 100

    exits  = [t for t in trades if t["type"] in ("EXIT","STOP")]
    pnls   = [t["pnl"] for t in exits if "pnl" in t]
    wins   = [p for p in pnls if p > 0]
    losses = [p for p in pnls if p <= 0]

    win_rate  = len(wins)/len(pnls)*100 if pnls else 0
    avg_win   = float(np.mean(wins))    if wins   else 0
    avg_loss  = float(np.mean(losses))  if losses else 0
    pf        = abs(sum(wins)/sum(losses)) if losses and sum(losses) != 0 else float("inf")
    ret_s     = pd.Series(eq).pct_change().dropna()
    sharpe    = (ret_s.mean()/ret_s.std())*np.sqrt(1560) if ret_s.std() > 0 else 0

    return {
        "final_equity":   round(eq[-1], 2),
        "total_return":   round(total_return, 2),
        "max_drawdown":   round(drawdown.min(), 2),
        "win_rate":       round(win_rate, 2),
        "avg_win":        round(avg_win, 2),
        "avg_loss":       round(avg_loss, 2),
        "profit_factor":  round(pf, 3),
        "sharpe":         round(sharpe, 3),
        "total_trades":   len(exits),
        "drawdown_series":drawdown,
        "equity_curve":   eq,
    }

# ══════════════════════════════════════════════════════════════════
# 3B. WALK-FORWARD OPTIMIZATION ENGINE
# ══════════════════════════════════════════════════════════════════

def run_walk_forward(df, cfg, progress_cb=None):
    """
    Walk-forward optimization:
    1. Split data into (train_bars + test_bars) folds
    2. For each fold: grid-search parameters on train window
    3. Apply best params to test window — record real out-of-sample result
    4. Return per-fold results + combined OOS equity curve

    progress_cb(pct, msg) — optional callback for UI progress bar
    """
    import itertools

    # ── Parse param grids ─────────────────────────────────────────
    def parse_range(key, default):
        raw = str(cfg.get(key, default))
        try:
            return [float(x.strip()) for x in raw.split(",") if x.strip()]
        except:
            return [float(default)]

    train_bars = int(cfg.get("wfo_train_bars", 1000))
    test_bars  = int(cfg.get("wfo_test_bars",  250))
    n_folds    = int(cfg.get("wfo_folds",      5))
    metric_key = str(cfg.get("wfo_metric",     "sharpe"))

    t1_vals    = [int(x) for x in parse_range("wfo_t1_range",    "35,40,45")]
    exit_vals  = [int(x) for x in parse_range("wfo_exit_range",  "48,50,52,55")]
    stop_vals  = parse_range("wfo_stop_range",  "3.0,4.0,5.0,6.0")
    trail_vals = parse_range("wfo_trail_range", "1.0,1.5,2.0")

    combos = list(itertools.product(t1_vals, exit_vals, stop_vals, trail_vals))
    total_steps = n_folds * len(combos)
    step = 0

    # ── Determine fold windows ────────────────────────────────────
    fold_size  = train_bars + test_bars
    total_bars = len(df)

    # Walk forward: anchor start, slide window
    folds = []
    for f in range(n_folds):
        train_start = f * test_bars
        train_end   = train_start + train_bars
        test_end    = train_end + test_bars
        if test_end > total_bars:
            break
        folds.append((train_start, train_end, test_end))

    if not folds:
        return None, "Not enough bars for walk-forward. Need at least " \
               f"{fold_size} bars, have {total_bars}."

    fold_results  = []
    oos_equity    = [float(cfg["initial_capital"])]
    oos_trades    = []
    running_eq    = float(cfg["initial_capital"])

    for fold_idx, (tr_s, tr_e, te_e) in enumerate(folds):
        df_train = df.iloc[tr_s:tr_e]
        df_test  = df.iloc[tr_e:te_e]

        # ── Grid search on training window ────────────────────────
        best_metric = -np.inf
        best_params = None

        for t1, exit_r, stop_p, trail_p in combos:
            step += 1
            if progress_cb:
                pct = int(step / total_steps * 100)
                progress_cb(pct, f"Fold {fold_idx+1}/{len(folds)} — "
                            f"testing T1={t1} Exit={exit_r} Stop={stop_p}%")

            # Skip invalid combos
            if t1 >= exit_r:
                continue
            t2 = max(t1 - 8, 20)
            t3 = max(t1 - 15, 15)

            trial_cfg = dict(cfg)
            trial_cfg.update({
                "t1_rsi":      t1,
                "t2_rsi":      t2,
                "t3_rsi":      t3,
                "exit_rsi":    exit_r,
                "stop_pct_t1": stop_p,
                "stop_pct_t2": max(stop_p - 1.0, 1.0),
                "stop_pct_t3": max(stop_p - 2.0, 1.0),
                "trail_perc":  trail_p,
            })

            try:
                eq_c, _, tr = run_backtest(df_train, trial_cfg)
                m = compute_metrics(eq_c, tr, cfg["initial_capital"])
            except Exception:
                continue

            # Score — pick metric
            score = {
                "sharpe":         m["sharpe"],
                "total_return":   m["total_return"],
                "profit_factor":  min(m["profit_factor"], 10),
                "win_rate":       m["win_rate"],
            }.get(metric_key, m["sharpe"])

            # Penalise if fewer than 3 trades — overfitted
            if m["total_trades"] < 3:
                score -= 999

            if score > best_metric:
                best_metric = score
                best_params = trial_cfg

        if best_params is None:
            best_params = dict(cfg)

        # ── Apply best params to test window ─────────────────────
        best_params["initial_capital"] = running_eq
        try:
            eq_oos, _, trades_oos = run_backtest(df_test, best_params)
            m_oos = compute_metrics(eq_oos, trades_oos, running_eq)
        except Exception as e:
            m_oos = {"total_return": 0, "sharpe": 0, "win_rate": 0,
                     "profit_factor": 0, "total_trades": 0,
                     "max_drawdown": 0, "final_equity": running_eq}
            trades_oos = []
            eq_oos     = [running_eq]

        # Chain equity curves
        oos_equity += list(eq_oos[1:])
        oos_trades += trades_oos
        running_eq  = eq_oos[-1]

        fold_results.append({
            "fold":          fold_idx + 1,
            "train_start":   df.index[tr_s],
            "train_end":     df.index[tr_e - 1],
            "test_start":    df.index[tr_e],

def run_futures_backtest(df, cfg):
    """
    Futures RSI reversion engine.
    Same logic as RSI Reversion but:
    - Position sized in contracts not equity %
    - P&L calculated as: points × multiplier × contracts
    - Stops defined in points not %
    - Tracks both point P&L and dollar P&L
    """
    equity       = float(cfg["initial_capital"])
    multiplier   = float(cfg.get("futures_multiplier", 5.0))
    t1_contracts = int(cfg.get("t1_contracts", 1))
    t2_contracts = int(cfg.get("t2_contracts", 1))
    t3_contracts = int(cfg.get("t3_contracts", 1))
    stop_pts_t1  = float(cfg.get("stop_pts_t1",  20.0))
    stop_pts_t2  = float(cfg.get("stop_pts_t2",  15.0))
    stop_pts_t3  = float(cfg.get("stop_pts_t3",  10.0))
    be_pts       = float(cfg.get("breakeven_pts",  8.0))
    trail_pts    = float(cfg.get("trail_pts",       5.0))
    t1_rsi       = cfg.get("t1_rsi",  40)
    t2_rsi       = cfg.get("t2_rsi",  32)
    t3_rsi       = cfg.get("t3_rsi",  25)
    ex_rsi       = cfg.get("exit_rsi", 50)
    cooldown     = int(cfg.get("cooldown_bars", 5))
    max_dl       = float(cfg.get("max_daily_loss", 3.0)) / 100
    margin_req   = float(cfg.get("futures_margin", 1200.0))

    closes   = df["Close"].values
    highs    = df["High"].values  if "High"  in df.columns else closes
    lows     = df["Low"].values   if "Low"   in df.columns else closes
    sma200   = df["sma200"].values if "sma200" in df.columns else np.full(len(closes), np.nan)
    sma50    = df["sma50"].values  if "sma50"  in df.columns else np.full(len(closes), np.nan)
    rsi_arr  = df["rsi"].values    if "rsi"    in df.columns else np.full(len(closes), 50.0)
    dates    = df.index

    # Pre-compute RSI crossunders
    xu_t1 = np.zeros(len(closes), bool)
    xu_t2 = np.zeros(len(closes), bool)
    xu_t3 = np.zeros(len(closes), bool)
    xo_ex = np.zeros(len(closes), bool)
    for i in range(1, len(closes)):
        if not (np.isnan(rsi_arr[i]) or np.isnan(rsi_arr[i-1])):
            xu_t1[i] = rsi_arr[i-1] >= t1_rsi and rsi_arr[i] < t1_rsi
            xu_t2[i] = rsi_arr[i-1] >= t2_rsi and rsi_arr[i] < t2_rsi
            xu_t3[i] = rsi_arr[i-1] >= t3_rsi and rsi_arr[i] < t3_rsi
            xo_ex[i] = rsi_arr[i-1] <= ex_rsi  and rsi_arr[i] > ex_rsi

    trades       = []
    equity_curve = []
    stop_curve   = []

    contracts    = 0       # current open contracts
    avg_entry    = 0.0
    trail_high   = 0.0
    open_tiers   = 0
    last_loss_bar = -999
    entry_date   = None
    daily_start  = equity
    current_day  = None
    profit_taken = False

    for i in range(1, len(closes)):
        close = float(closes[i])
        date  = dates[i]
        day   = date.date() if hasattr(date, "date") else date

        if day != current_day:
            daily_start = equity
            current_day = day

        above_200  = not np.isnan(sma200[i]) and close > sma200[i]
        above_50   = not np.isnan(sma50[i])  and close > sma50[i]
        cool_ok    = (i - last_loss_bar) >= cooldown
        daily_ok   = (daily_start - equity) <= daily_start * max_dl
        can_enter  = above_200 and cool_ok and daily_ok

        # ── Stop check ───────────────────────────────────────────
        if contracts > 0:
            trail_high = max(trail_high, close)
            stop_pts   = [0, stop_pts_t1, stop_pts_t2, stop_pts_t3][min(open_tiers, 3)]
            base_stop  = avg_entry - stop_pts

            # Breakeven + trailing
            if close >= avg_entry + be_pts:
                trail_stop = trail_high - trail_pts
                base_stop  = max(avg_entry, trail_stop, base_stop)

            if close <= base_stop:
                pts = (close - avg_entry)
                pnl = pts * multiplier * contracts
                equity += pnl
                trades.append({
                    "type": "STOP", "date": date, "entry_date": entry_date,
                    "entry": avg_entry, "exit": close,
                    "qty": contracts, "pnl": pnl,
                    "points": round(pts, 2), "bar": i,
                })
                if pnl < 0: last_loss_bar = i
                contracts = 0; avg_entry = 0.0
                trail_high = 0.0; open_tiers = 0
                profit_taken = False; entry_date = None

        # ── Entry signals ─────────────────────────────────────────
        if can_enter:
            # Check margin availability
            margin_used = contracts * margin_req
            margin_avail = equity - margin_used

            if xu_t1[i] and contracts == 0:
                if margin_avail >= margin_req * t1_contracts:
                    contracts  = t1_contracts
                    avg_entry  = close
                    trail_high = close
                    open_tiers = 1
                    entry_date = date
                    trades.append({
                        "type": "T1_ENTRY", "date": date,
                        "price": close, "bar": i, "contracts": contracts
                    })

            elif contracts > 0 and xu_t2[i] and open_tiers == 1:
                if margin_avail >= margin_req * t2_contracts:
                    new_avg  = (contracts*avg_entry + t2_contracts*close) / (contracts+t2_contracts)
                    contracts += t2_contracts
                    avg_entry  = new_avg
                    open_tiers = 2
                    trades.append({
                        "type": "T2", "date": date,
                        "price": close, "bar": i, "contracts": contracts
                    })

            elif contracts > 0 and xu_t3[i] and open_tiers == 2:
                if margin_avail >= margin_req * t3_contracts:
                    new_avg  = (contracts*avg_entry + t3_contracts*close) / (contracts+t3_contracts)
                    contracts += t3_contracts
                    avg_entry  = new_avg
                    open_tiers = 3
                    trades.append({
                        "type": "T3", "date": date,
                        "price": close, "bar": i, "contracts": contracts
                    })

        # ── Exit signal ───────────────────────────────────────────
        if contracts > 0:
            exit_trig = xo_ex[i] or above_50
            if exit_trig and not profit_taken:
                pts = (close - avg_entry)
                pnl = pts * multiplier * contracts
                equity += pnl
                trades.append({
                    "type": "EXIT", "date": date, "entry_date": entry_date,
                    "entry": avg_entry, "exit": close,
                    "qty": contracts, "pnl": pnl,
                    "points": round(pts, 2), "bar": i,
                })
                contracts  = 0; avg_entry = 0.0
                trail_high = 0.0; open_tiers = 0
                profit_taken = False; entry_date = None

        mtm = equity + (close - avg_entry) * multiplier * contracts if contracts > 0 else equity
        equity_curve.append(mtm)
        pts_stop = avg_entry - [0, stop_pts_t1, stop_pts_t2, stop_pts_t3][min(open_tiers,3)]
        stop_curve.append(pts_stop if contracts > 0 else np.nan)

    return equity_curve, stop_curve, trades

# ══════════════════════════════════════════════════════════════════
    equity       = float(cfg["initial_capital"])
    leverage     = float(cfg["leverage"])
    gap_min      = cfg["gap_min_pct"]    / 100
    gap_max      = cfg["gap_max_pct"]    / 100
    stop_pct     = cfg["gap_stop_pct"]   / 100
    target_pct   = cfg["gap_target_pct"] / 100
    max_hold     = int(cfg["gap_max_hold_bars"])
    vol_mult     = cfg["gap_min_volume"]
    direction    = cfg["gap_direction"]   # "up", "down", "both"
    max_daily_loss = cfg["max_daily_loss"] / 100

    trades       = []
    equity_curve = []
    stop_curve   = []

    closes   = df["Close"].values
    opens    = df["Open"].values  if "Open"   in df.columns else closes
    highs    = df["High"].values  if "High"   in df.columns else closes
    lows     = df["Low"].values   if "Low"    in df.columns else closes
    volumes  = df["Volume"].values
    vol_sma  = df["vol_sma"].values
    dates    = df.index

    pos_size     = 0.0
    entry_price  = 0.0
    pos_dir      = 0      # +1 long, -1 short
    entry_bar    = -1
    stop_level   = 0.0
    target_level = 0.0
    daily_start  = equity
    current_day  = None

    for i in range(1, len(df)):
        close  = float(closes[i])
        open_  = float(opens[i])
        high   = float(highs[i])
        low    = float(lows[i])
        volume = float(volumes[i])
        prev_close = float(closes[i-1])
        date   = dates[i]
        day    = date.date() if hasattr(date, "date") else date

        # Daily reset
        if day != current_day:
            daily_start = equity
            current_day = day

        daily_loss_ok = (daily_start - equity) <= daily_start * max_daily_loss

        # ── Exit logic (check first on each bar) ──────────────────
        if pos_size > 0:
            bars_held = i - entry_bar
            closed    = False
            exit_px   = close
            exit_type = ""

            if pos_dir == 1:   # long fade
                if low <= stop_level:
                    exit_px = stop_level; exit_type = "STOP"; closed = True
                elif high >= target_level:
                    exit_px = target_level; exit_type = "TARGET"; closed = True
                elif bars_held >= max_hold:
                    exit_type = "TIMEOUT"; closed = True
            else:              # short fade
                if high >= stop_level:
                    exit_px = stop_level; exit_type = "STOP"; closed = True
                elif low <= target_level:
                    exit_px = target_level; exit_type = "TARGET"; closed = True
                elif bars_held >= max_hold:
                    exit_type = "TIMEOUT"; closed = True

            if closed:
                if pos_dir == 1:
                    pnl = (exit_px - entry_price) * pos_size
                else:
                    pnl = (entry_price - exit_px) * pos_size
                equity += pnl
                trades.append({
                    "type":  exit_type,
                    "date":  date,
                    "entry": entry_price,
                    "exit":  exit_px,
                    "qty":   pos_size,
                    "pnl":   pnl,
                    "bar":   i,
                    "dir":   "LONG" if pos_dir == 1 else "SHORT",
                })
                pos_size = 0.0; pos_dir = 0; entry_bar = -1
                equity_curve.append(equity); stop_curve.append(np.nan)
                continue

        # ── Entry logic — only on first bar of day (9:30 open) ────
        is_open_bar = False
        if hasattr(date, "hour"):
            is_open_bar = (date.hour == 9 and date.minute == 30)
        elif i > 0:
            prev_day = dates[i-1].date() if hasattr(dates[i-1], "date") else dates[i-1]
            is_open_bar = (day != prev_day)

        if pos_size == 0 and daily_loss_ok and is_open_bar:
            gap_pct = (open_ - prev_close) / prev_close

            # Volume check
            vol_ok = (volume > vol_sma[i] * vol_mult) if not np.isnan(vol_sma[i]) else True

            # Gap up → short fade
            if (direction in ("up","both") and
                    gap_pct >= gap_min and gap_pct <= gap_max and vol_ok):
                qty         = (equity * leverage) / open_
                pos_size    = qty
                pos_dir     = -1
                entry_price = open_
                entry_bar   = i
                stop_level  = open_ * (1 + stop_pct)
                target_level= open_ * (1 - target_pct)
                trades.append({
                    "type":"GAP_UP_SHORT","date":date,
                    "price":open_,"gap_pct":round(gap_pct*100,2),"bar":i
                })

            # Gap down → long fade
            elif (direction in ("down","both") and
                      gap_pct <= -gap_min and abs(gap_pct) <= gap_max and vol_ok):
                qty         = (equity * leverage) / open_
                pos_size    = qty
                pos_dir     = 1
                entry_price = open_
                entry_bar   = i
                stop_level  = open_ * (1 - stop_pct)
                target_level= open_ * (1 + target_pct)
                trades.append({
                    "type":"GAP_DN_LONG","date":date,
                    "price":open_,"gap_pct":round(gap_pct*100,2),"bar":i
                })

        # Mark-to-market
        if pos_size > 0:
            if pos_dir == 1:
                mtm = equity + (close - entry_price) * pos_size
            else:
                mtm = equity + (entry_price - close) * pos_size
            stop_curve.append(stop_level)
        else:
            mtm = equity
            stop_curve.append(np.nan)
        equity_curve.append(mtm)

    return equity_curve, stop_curve, trades

def now_et():
    if _USE_PYTZ:
        return datetime.datetime.now(_TZ_ET)
    return datetime.datetime.now(datetime.timezone(datetime.timedelta(hours=-4)))

def is_market_open():
    n = now_et()
    if n.weekday() >= 5: return False
    o = n.replace(hour=9,  minute=30, second=0, microsecond=0)
    c = n.replace(hour=16, minute=0,  second=0, microsecond=0)
    return o <= n <= c

def secs_to_next_hour():
    n   = datetime.datetime.now()
    nxt = n.replace(minute=0, second=0, microsecond=0) + datetime.timedelta(hours=1)
    return int((nxt - n).total_seconds())

def max_period_for_interval(interval):
    """yfinance intraday data limits by interval."""
    return {
        "1m":  "7d",
        "2m":  "60d",
        "5m":  "60d",
        "15m": "60d",
        "30m": "60d",
        "60m": "730d",
        "1h":  "730d",
        "90m": "60d",
        "1d":  "730d",
        "1wk": "730d",
        "1mo": "730d",
    }.get(interval, "60d")


class DataFetcher(threading.Thread):
    def __init__(self, q, cfg):
        super().__init__(daemon=True)
        self.q = q; self.cfg = cfg
        self._stop = threading.Event()

    def stop(self): self._stop.set()

    def run(self):
        while not self._stop.is_set():
            try:
                interval = self.cfg.get("interval", "1h")
                period   = max_period_for_interval(interval)
                df = yf.download(self.cfg["ticker"],
                                 period=period,
                                 interval=interval,
                                 auto_adjust=True, progress=False)
                if df is not None and len(df) > 10:
                    if isinstance(df.columns, pd.MultiIndex):
                        df.columns = [c[0] for c in df.columns]
                    else:
                        df.columns = [c[0] if isinstance(c, tuple) else c
                                      for c in df.columns]
                    df = df.loc[:, ~df.columns.duplicated()]
                    if isinstance(df["Close"], pd.DataFrame):
                        df["Close"] = df["Close"].iloc[:, 0]
                    df = build_indicators(df)
                    df.dropna(subset=["rsi","sma50"], inplace=True)
                    self.q.put(("data", df))
            except Exception as e:
                self.q.put(("error", str(e)))
            for _ in range(self.cfg["refresh_seconds"]):
                if self._stop.is_set(): break
                time.sleep(1)


class LiveDataFetcher(threading.Thread):
    """Fetches data for a single ticker and tags results with the symbol
    so the multi-ticker dashboard knows which card to update."""
    def __init__(self, q, cfg):
        super().__init__(daemon=True)
        self.q = q; self.cfg = cfg
        self._stop = threading.Event()

    def stop(self): self._stop.set()

    def run(self):
        sym = self.cfg["ticker"]
        while not self._stop.is_set():
            try:
                interval = self.cfg.get("interval", "1h")
                period   = max_period_for_interval(interval)
                df = yf.download(sym,
                                 period=period,
                                 interval=interval,
                                 auto_adjust=True, progress=False)
                if df is None or len(df) < 10:
                    self.q.put(("error", f"{sym}: no data returned"))
                else:
                    # Flatten ALL forms of multi-level columns yfinance can produce
                    if isinstance(df.columns, pd.MultiIndex):
                        df.columns = [c[0] for c in df.columns]
                    else:
                        df.columns = [c[0] if isinstance(c, tuple) else c
                                      for c in df.columns]
                    # Drop duplicate columns if any
                    df = df.loc[:, ~df.columns.duplicated()]
                    # Ensure Close is a plain Series
                    if isinstance(df["Close"], pd.DataFrame):
                        df["Close"] = df["Close"].iloc[:, 0]

                    df = build_indicators(df)
                    df.dropna(subset=["rsi", "sma50"], inplace=True)
                    if len(df) > 10:
                        self.q.put(("data", (sym, df)))
                    else:
                        self.q.put(("error", f"{sym}: insufficient bars after indicators"))
            except Exception as e:
                self.q.put(("error", f"{sym}: {e}"))
            for _ in range(60):
                if self._stop.is_set(): break
                time.sleep(1)

    def _ld_add_nasdaq(self):
        """Stagger-load the Nasdaq 100, then auto-prune blocked tickers."""
        NDX100 = [
            "MSFT","AAPL","NVDA","AMZN","META","GOOGL","GOOG","TSLA","AVGO","COST",
            "NFLX","ASML","AMD","PEP","LIN","QCOM","ADBE","INTU","AMAT","TXN",
            "CSCO","AMGN","BKNG","MU","ISRG","HON","REGN","LRCX","ADI",
            "PANW","VRTX","KLAC","SBUX","MELI","GILD","INTC","MDLZ","SNPS","CDNS",
            "CTAS","CEG","MRVL","CRWD","ADP","CSX","NXPI","PYPL","FTNT","WDAY",
            "ADSK","TEAM","ROP","ABNB","DXCM","ROST","MNST","PAYX","FAST","PCAR",
            "KDP","EXC","ODFL","BKR","CPRT","VRSK","CTSH","ON","IDXX",
            "DDOG","FANG","GEHC","BIIB","MCHP","EA","XEL","ZS","TTWO",
            "CDW","DLTR","ILMN","MRNA","ALGN","RIVN","ZM","OKTA",
            "MTCH","ENPH","CSGP","CHTR","LULU",
        ]
        to_add = [s for s in NDX100 if s not in self._ld_tickers]
        if not to_add:
            self._ld_log_msg("All Nasdaq 100 tickers already loaded", "info")
            return

        batch_size = 5
        delay_ms   = 700

        self._ld_log_msg(f"Loading {len(to_add)} Nasdaq tickers — staggering to avoid rate limits...", "info")

        def load_batch(batch, batch_num):
            for sym in batch:
                if sym not in self._ld_tickers:
                    self._ld_build_card(sym)
                    cfg = dict(CONFIG)
                    cfg["ticker"] = sym
                    fetcher = LiveDataFetcher(self._ld_queue, cfg)
                    fetcher.start()
                    self._ld_tickers[sym].update({
                        "fetcher":   fetcher,
                        "df":        None,
                        "pos":       0.0, "entry": 0.0, "tier": 0,
                        "trail_hi":  0.0,
                        "equity":    float(CONFIG["initial_capital"]),
                        "wins":      0, "losses": 0,
                        "bar_count": 0, "last_loss": -999, "prev_rsi": None,
                    })
            loaded = min((batch_num + 1) * batch_size, len(to_add))
            self._ld_log_msg(f"Queued {loaded}/{len(to_add)} tickers...", "info")

        for i in range(0, len(to_add), batch_size):
            batch     = to_add[i:i+batch_size]
            batch_num = i // batch_size
            self.after(batch_num * delay_ms,
                       lambda b=batch, n=batch_num: load_batch(b, n))

        # After all batches loaded + 90s for data to arrive, auto-prune blocked
        total_batches = (len(to_add) + batch_size - 1) // batch_size
        prune_delay   = total_batches * delay_ms + 90_000  # 90s for data to arrive
        self.after(prune_delay, self._ld_prune_blocked)
        self._ld_log_msg(
            f"Auto-prune will run in ~{prune_delay//1000}s — removing blocked tickers", "info")

    def _ld_prune_blocked(self):
        """Remove tickers that are blocked or have no data — keeps dashboard fast."""
        if not hasattr(self, "_ld_tickers"): return

        to_remove = []
        for sym, data in list(self._ld_tickers.items()):
            # Never remove tickers with open positions
            if data.get("tier", 0) > 0:
                continue
            df = data.get("df")
            # Remove if no data arrived
            if df is None or len(df) < 2:
                to_remove.append((sym, "no data"))
                continue
            # Remove if blocked — price below SMA 200
            try:
                last     = df.iloc[-1]
                close    = float(last["Close"])
                sma200   = float(last["sma200"]) if not pd.isna(last["sma200"]) else None
                sma50    = float(last["sma50"])  if not pd.isna(last["sma50"])  else None
                above200 = sma200 is not None and close > sma200

                # Check slope filters
                slope200_on = str(CONFIG.get("sma200_slope_on","false")).lower() not in ("false","0","no","off")
                slope50_on  = str(CONFIG.get("sma50_slope_on","false")).lower()  not in ("false","0","no","off")
                slope_ok    = True
                if slope200_on and len(df) > 10:
                    s200 = df["sma200"].values
                    if not pd.isna(s200[-1]) and not pd.isna(s200[-10]):
                        slope_ok = (s200[-1] - s200[-10]) / s200[-10] * 100 >= CONFIG.get("sma200_slope_min", 0.0)
                s50_ok = True
                if slope50_on and len(df) > 5:
                    s50 = df["sma50"].values
                    if not pd.isna(s50[-1]) and not pd.isna(s50[-5]):
                        s50_ok = (s50[-1] - s50[-5]) / s50[-5] * 100 >= CONFIG.get("sma50_slope_min", 0.0)

                can_trade = above200 and slope_ok and s50_ok
                if not can_trade:
                    to_remove.append((sym, "blocked"))
            except Exception:
                to_remove.append((sym, "error"))

        for sym, reason in to_remove:
            data  = self._ld_tickers.get(sym)
            if not data: continue
            frame = data.get("card_frame")
            if "fetcher" in data:
                data["fetcher"].stop()
            self._ld_tickers.pop(sym, None)
            if frame:
                try: frame.destroy()
                except: pass

        kept = len(self._ld_tickers)
        self._ld_log_msg(
            f"Auto-prune complete — removed {len(to_remove)} blocked/no-data tickers  "
            f"|  {kept} tradeable tickers remaining", "info")

            "prev_rsi":  None,
        })
        self._ld_log_msg(f"Added {sym} — fetching data...", "info")
        self._ld_ticker_entry.delete(0, "end")

    def _ld_remove_ticker(self, sym, card_frame=None):
        if sym not in self._ld_tickers: return
        data = self._ld_tickers.pop(sym)

        # Stop the fetcher thread
        if "fetcher" in data:
            data["fetcher"].stop()

        # Destroy just this card — no rebuild, others unaffected
        frame = card_frame or data.get("card_frame")
        if frame:
            try:
                frame.destroy()
            except Exception:
                pass

        self._ld_log_msg(f"Removed {sym}", "info")

    def _ld_render_chart(self, win, sym, data, df):
        """Draw the live chart — last 120 bars with entry/stop overlays."""
        if not win.winfo_exists(): return
        ax_price = win._ax_price
        ax_rsi   = win._ax_rsi
        canvas   = win._canvas

        ax_price.cla()
        ax_rsi.cla()

        for ax in [ax_price, ax_rsi]:
            ax.set_facecolor(S["panel"])
            ax.tick_params(colors=S["subtext"], labelsize=8)
            for sp in ax.spines.values():
                sp.set_edgecolor(S["border"])

        n_bars = 120
        df_c   = df.iloc[-n_bars:].copy()
        dates  = df_c.index
        closes = df_c["Close"].values
        opens  = df_c["Open"].values  if "Open"  in df_c.columns else closes
        highs  = df_c["High"].values  if "High"  in df_c.columns else closes
        lows   = df_c["Low"].values   if "Low"   in df_c.columns else closes
        rsi_v  = df_c["rsi"].values   if "rsi"   in df_c.columns else np.full(len(closes), 50)
        sma200 = df_c["sma200"].values if "sma200" in df_c.columns else None
        sma50  = df_c["sma50"].values  if "sma50"  in df_c.columns else None

        x  = np.arange(len(df_c))
        cw = 0.6

        # ── Candles — dim outside open position ───────────────────
        in_pos   = data.get("pos", 0) > 0
        entry_px = data.get("entry", 0.0)
        tier     = data.get("tier", 0)

        for i in range(len(df_c)):
            o, h, l, c = opens[i], highs[i], lows[i], closes[i]
            if any(np.isnan([o,h,l,c])): continue
            bull  = c >= o
            col   = S["green"] if bull else S["red"]
            alpha = 0.9
            ax_price.plot([i,i],[l,h], color=col, lw=0.8, alpha=alpha, zorder=2)
            ax_price.add_patch(mpatches.Rectangle(
                (i-cw/2, min(o,c)), cw, max(abs(c-o), cw*0.1),
                facecolor=col, edgecolor=col, lw=0, zorder=3, alpha=alpha))

        # SMAs
        if sma200 is not None:
            ax_price.plot(x, sma200, color=S["blue"],
                          lw=1.2, alpha=0.8, label="SMA 200", zorder=4)
        if sma50 is not None:
            ax_price.plot(x, sma50, color=S["orange"],
                          lw=1.0, alpha=0.8, label="SMA 50", zorder=4)

        # ── Trade overlays if in position ─────────────────────────
        live_pnl = 0.0
        if in_pos and entry_px > 0:
            cur      = float(closes[-1])
            live_pnl = (cur - entry_px) * data["pos"]
            pnl_col  = S["green"] if live_pnl >= 0 else S["red"]

            # Entry horizontal line
            ax_price.axhline(entry_px, color=S["orange"],
                             lw=1.2, linestyle="--", alpha=0.9,
                             label=f"Entry ${entry_px:.2f}")

            # Stop level
            sp_pct  = [0, CONFIG["stop_pct_t1"], CONFIG["stop_pct_t2"],
                       CONFIG["stop_pct_t3"]][min(tier, 3)] / 100
            stop_px = entry_px * (1 - sp_pct)
            ax_price.axhline(stop_px, color=S["red"],
                             lw=0.9, linestyle=":", alpha=0.8,
                             label=f"Stop ${stop_px:.2f}")

            # Shade between entry and current
            ax_price.fill_between(x, entry_px, closes,
                                  color=pnl_col, alpha=0.06)

            # Entry marker on the bar where position was opened
            # (approximate — mark last bar as current)
            ax_price.scatter([len(df_c)-1], [entry_px * 0.996],
                             marker="^", color=S["orange"],
                             s=140, zorder=8, edgecolors=S["bg"])

            # P&L annotation
            pnl_str = f"{'+' if live_pnl>=0 else ''}${live_pnl:.2f}"
            ax_price.text(len(df_c)-2, (entry_px+cur)/2,
                         f" {pnl_str}",
                         color=pnl_col, fontsize=9,
                         fontweight="bold", va="center", ha="right")

            # Tier label
            tier_names = ["","T1","T1+T2","FULLY LOADED"]
            ax_price.text(1, entry_px * 1.001,
                         f" {tier_names[min(tier,3)]}",
                         color=S["orange"], fontsize=8, va="bottom")

        # Current price line
        ax_price.axhline(float(closes[-1]),
                         color=S["subtext"], lw=0.5,
                         linestyle=":", alpha=0.5)

        # ── Chart labels ──────────────────────────────────────────
        pad = (np.nanmax(highs) - np.nanmin(lows)) * 0.05
        ax_price.set_ylim(np.nanmin(lows)-pad, np.nanmax(highs)+pad)
        ax_price.set_xlim(-1, len(df_c))
        ax_price.set_ylabel("Price", color=S["subtext"], fontsize=8)
        ax_price.legend(loc="upper left", fontsize=6,
                        facecolor=S["bg"], labelcolor=S["text"],
                        framealpha=0.85, edgecolor=S["border"], ncol=4)

        tier_names2 = ["NONE","IN T1","T1+T2","FULLY LOADED"]
        pos_cols    = [S["subtext"],S["orange"],S["red"],S["purple"]]
        tc = pos_cols[min(tier,3)]
        pnl_part = f"  |  Live P&L: {'+' if live_pnl>=0 else ''}${live_pnl:.2f}" if in_pos else ""
        ax_price.set_title(
            f"{sym}  |  Position: {tier_names2[min(tier,3)]}{pnl_part}"
            f"  |  RSI: {rsi_v[-1]:.1f}"
            f"  |  ${float(closes[-1]):.2f}"
            f"  |  Updated: {datetime.datetime.now().strftime('%H:%M:%S')}",
            color=tc, fontsize=9, pad=5)

        # ── RSI panel ─────────────────────────────────────────────
        ax_rsi.plot(x, rsi_v, color=S["purple"], lw=0.9)
        for lvl, col in [(CONFIG["t1_rsi"],S["orange"]),
                          (CONFIG["t2_rsi"],S["red"]),
                          (CONFIG["t3_rsi"],S["purple"]),
                          (CONFIG["exit_rsi"],S["green"])]:
            ax_rsi.axhline(lvl, color=col, lw=0.5, linestyle="--", alpha=0.7)
        ax_rsi.fill_between(x, 0, rsi_v,
                             where=rsi_v < CONFIG["t3_rsi"],
                             color=S["purple"], alpha=0.2)
        ax_rsi.set_ylim(0, 100)
        ax_rsi.set_xlim(-1, len(df_c))
        ax_rsi.set_ylabel("RSI", color=S["subtext"], fontsize=8)

        # X axis date labels
        n_ticks   = min(10, len(df_c))
        tick_locs = np.linspace(0, len(df_c)-1, n_ticks, dtype=int)
        tick_lbls = []
        for i in tick_locs:
            try:
                d = dates[i]
                tick_lbls.append(d.strftime("%m/%d %H:%M")
                                  if hasattr(d,"strftime") else str(d)[:10])
            except:
                tick_lbls.append("")
        ax_rsi.set_xticks(tick_locs)
        ax_rsi.set_xticklabels(tick_lbls, fontsize=7.5,
                                color=S["text"], rotation=0)
        ax_price.set_xticks([])
        for ax in [ax_price, ax_rsi]:
            ax.yaxis.grid(True, color=S["border"], lw=0.3, alpha=0.4)
            for loc in tick_locs:
                ax.axvline(loc, color=S["border"], lw=0.3, alpha=0.3)

        canvas.draw_idle()

    def _ld_force_exit(self, sym):
        if sym not in self._ld_tickers: return
        data = self._ld_tickers[sym]
        if data.get("pos", 0) <= 0:
            self._ld_log_msg(f"{sym}: no open position", "info")
            return
        df = data.get("df")
        if df is None: return
        close = float(df.iloc[-1]["Close"])
        pnl   = (close - data["entry"]) * data["pos"]
        data["equity"] += pnl
        data["pos"]    = 0.0
        data["entry"]  = 0.0
        data["tier"]   = 0
        data["trail_hi"] = 0.0
        if pnl >= 0: data["wins"]   += 1
        else:        data["losses"] += 1
        self._ld_session_pnl += pnl
        pnl_str = f"+${pnl:.2f}" if pnl >= 0 else f"${pnl:.2f}"
        self._ld_log_msg(f"[{sym}] FORCE EXIT @ ${close:.2f}  P&L: {pnl_str}", "exit")

    def _ld_clear_all(self):
        for sym, data in list(self._ld_tickers.items()):
            if "fetcher" in data:
                data["fetcher"].stop()
        self._ld_tickers.clear()
        for w in self._ld_grid.winfo_children():
            w.destroy()
        self._ld_log_msg("All tickers removed", "info")

    def _ld_tick_main(self):
        """Per-second update loop — runs in main window event loop."""
        if not self._ld_running:
            return

        try:
            while True:
                kind, payload = self._ld_queue.get_nowait()
                if kind == "data":
                    sym, df = payload
                    if sym in self._ld_tickers:
                        self._ld_tickers[sym]["df"] = df
                        self._ld_process_ticker(sym, df)
                elif kind == "error":
                    self._ld_log_msg(f"Data error: {payload}", "warn")
        except queue.Empty:
            pass

        # Update session header
        if hasattr(self, "_ld_session_lbl"):
            col = S["green"] if self._ld_session_pnl >= 0 else S["red"]
            sign = "+" if self._ld_session_pnl >= 0 else ""
            self._ld_session_lbl.config(
                text=f"Session P&L: {sign}${self._ld_session_pnl:.2f}"
                     f"  |  Wins: {self._ld_session_wins}"
                     f"  |  Losses: {self._ld_session_losses}",
                fg=col)

        # Re-sort cards every 5 seconds
        self._ld_sort_counter = getattr(self, "_ld_sort_counter", 0) + 1
        if self._ld_sort_counter >= 5:
            self._ld_sort_counter = 0
            self._ld_sort_cards()

        self.after(1000, self._ld_tick_main)

    def _ld_sort_cards(self):
        """Re-order cards: IN POSITION → WATCHING T1 → rest (by original load order)."""
        if not hasattr(self, "_ld_grid"): return

        # NDX100 original order for fallback sort
        NDX_ORDER = [
            "VOO",
            "MSFT","AAPL","NVDA","AMZN","META","GOOGL","GOOG","TSLA","AVGO","COST",
            "NFLX","ASML","AMD","PEP","LIN","QCOM","ADBE","INTU","AMAT","TXN",
            "CSCO","AMGN","BKNG","MU","ISRG","HON","REGN","LRCX","ADI",
            "PANW","VRTX","KLAC","SBUX","MELI","GILD","INTC","MDLZ","SNPS","CDNS",
            "CTAS","CEG","MRVL","CRWD","ADP","CSX","NXPI","PYPL","FTNT","WDAY",
            "ADSK","TEAM","ROP","ABNB","DXCM","ROST","MNST","PAYX","FAST","PCAR",
            "KDP","EXC","ODFL","BKR","CPRT","VRSK","CTSH","ON","IDXX",
            "DDOG","FANG","GEHC","BIIB","MCHP","EA","XEL","ZS","TTWO",
            "CDW","DLTR","ILMN","MRNA","ALGN","RIVN","ZM","OKTA",
            "MTCH","ENPH","CSGP","CHTR","LULU",
        ]
        ndx_rank = {sym: i for i, sym in enumerate(NDX_ORDER)}

        def sort_key(sym):
            data = self._ld_tickers.get(sym, {})
            tier = data.get("tier", 0)
            # Priority 0 = in position (tier > 0)
            # Priority 1 = watching T1 (has data, tier=0, can_enter, rsi approaching)
            # Priority 2 = blocked / loading
            if tier > 0:
                priority = 0
            else:
                df = data.get("df")
                if df is not None and len(df) > 1:
                    try:
                        rsi = float(df.iloc[-1]["rsi"])
                        t1  = CONFIG.get("t1_rsi", 40)
                        # "Watching" = has data, rsi within 15 points of T1
                        priority = 1 if rsi < t1 + 15 else 2
                    except:
                        priority = 2
                else:
                    priority = 3   # still loading
            rank = ndx_rank.get(sym, 999)
            return (priority, rank)

        syms_sorted = sorted(self._ld_tickers.keys(), key=sort_key)

        # Re-pack card frames in sorted order
        for sym in syms_sorted:
            data = self._ld_tickers.get(sym, {})
            frame = data.get("card_frame")
            if frame and frame.winfo_exists():
                frame.pack_forget()
                # Also remove the separator line below it
                frame.pack(fill="x", padx=6, pady=4)

    def _ld_process_ticker(self, sym, df):
        """Run live signal logic for one ticker and update its card."""
        data = self._ld_tickers.get(sym)
        if not data or "widgets" not in data: return
        if df is None or len(df) < 2: return

        # Safe defaults — handles race where fetcher returns before .update() runs
        data.setdefault("pos",       0.0)
        data.setdefault("entry",     0.0)
        data.setdefault("tier",      0)
        data.setdefault("trail_hi",  0.0)
        data.setdefault("equity",    float(CONFIG["initial_capital"]))
        data.setdefault("wins",      0)
        data.setdefault("losses",    0)
        data.setdefault("bar_count", 0)
        data.setdefault("last_loss", -999)
        data.setdefault("prev_rsi",  None)

        w = data["widgets"]

        last     = df.iloc[-1]
        prev     = df.iloc[-2]
        close    = float(last["Close"])
        rsi      = float(last["rsi"])    if not pd.isna(last["rsi"])    else None
        sma200_v = float(last["sma200"]) if not pd.isna(last["sma200"]) else None
        sma50_v  = float(last["sma50"])  if not pd.isna(last["sma50"])  else None
        prev_rsi = float(prev["rsi"])    if not pd.isna(prev["rsi"])    else None

        t1_rsi = CONFIG["t1_rsi"]
        t2_rsi = CONFIG["t2_rsi"]
        t3_rsi = CONFIG["t3_rsi"]
        ex_rsi = CONFIG["exit_rsi"]
        lev    = float(CONFIG["leverage"])

        above_200 = sma200_v is not None and close > sma200_v
        above_50  = sma50_v  is not None and close > sma50_v
        cool_ok   = data["bar_count"] - data["last_loss"] >= CONFIG["cooldown_bars"]

        # SMA 200 slope filter
        slope_on  = str(CONFIG.get("sma200_slope_on","true")).lower() not in ("false","0","no","off")
        slope_bars = int(CONFIG.get("sma200_slope_bars", 10))
        slope_min  = float(CONFIG.get("sma200_slope_min", 0.0))
        if slope_on and len(df) > slope_bars + 1:
            sma200_arr = df["sma200"].values
            if not np.isnan(sma200_arr[-1]) and not np.isnan(sma200_arr[-slope_bars-1]):
                prev_val = sma200_arr[-slope_bars-1]
                slope_pct = (sma200_arr[-1] - prev_val) / prev_val * 100 if prev_val != 0 else 0
                slope_ok = slope_pct >= slope_min
            else:
                slope_ok = True
        else:
            slope_ok = True

        s50_on   = str(CONFIG.get("sma50_slope_on","true")).lower() not in ("false","0","no","off")
        s50_bars = int(CONFIG.get("sma50_slope_bars", 5))
        s50_min  = float(CONFIG.get("sma50_slope_min", 0.0))
        if s50_on and len(df) > s50_bars + 1:
            sma50_arr = df["sma50"].values
            if not np.isnan(sma50_arr[-1]) and not np.isnan(sma50_arr[-s50_bars-1]):
                prev50 = sma50_arr[-s50_bars-1]
                s50_pct = (sma50_arr[-1] - prev50) / prev50 * 100 if prev50 != 0 else 0
                s50_slope_ok = s50_pct >= s50_min
            else:
                s50_slope_ok = True
        else:
            s50_slope_ok = True

        can_enter = above_200 and cool_ok and slope_ok and s50_slope_ok

        xu_t1 = prev_rsi and rsi and prev_rsi >= t1_rsi and rsi < t1_rsi
        xu_t2 = prev_rsi and rsi and prev_rsi >= t2_rsi and rsi < t2_rsi
        xu_t3 = prev_rsi and rsi and prev_rsi >= t3_rsi and rsi < t3_rsi
        xo_ex = prev_rsi and rsi and prev_rsi <= ex_rsi  and rsi > ex_rsi

        # ── Stop check ────────────────────────────────────────────
        if data["pos"] > 0 and data["entry"] > 0:
            data["trail_hi"] = max(data["trail_hi"], close)
            sp = [0, CONFIG["stop_pct_t1"],
                  CONFIG["stop_pct_t2"],
                  CONFIG["stop_pct_t3"]][min(data["tier"],3)] / 100
            base_stop  = data["entry"] * (1 - sp)
            profit_pct = (close - data["entry"]) / data["entry"]
            be_pct     = CONFIG["breakeven_pct"] / 100
            if profit_pct > be_pct:
                trail = data["trail_hi"] * (1 - CONFIG["trail_perc"] / 100)
                base_stop = max(data["entry"], trail, base_stop)
            if close <= base_stop:
                pnl = (close - data["entry"]) * data["pos"]
                data["equity"] += pnl
                data["pos"]  = 0.0; data["entry"]  = 0.0
                data["tier"] = 0;   data["trail_hi"] = 0.0
                if pnl < 0:
                    data["losses"]    += 1
                    data["last_loss"]  = data["bar_count"]
                    self._ld_session_losses += 1
                else:
                    data["wins"] += 1
                    self._ld_session_wins += 1
                self._ld_session_pnl += pnl
                pnl_str = f"+${pnl:.2f}" if pnl >= 0 else f"${pnl:.2f}"
                self._ld_log_msg(
                    f"[{sym}] ⛔ STOP HIT @ ${close:.2f}  P&L: {pnl_str}  "
                    f"Equity: ${data['equity']:.2f}", "stop")
                notify(f"Rush Algo's — {sym} STOP",
                       f"{sym} stop hit @ ${close:.2f}  P&L: {pnl_str}")

        # ── Entry signals ─────────────────────────────────────────
        if can_enter:
            if xu_t1 and data["tier"] == 0:
                qty = (data["equity"] * CONFIG.get("t1_size",1.0) * lev) / close
                data["pos"]   = qty; data["entry"]  = close
                data["tier"]  = 1;   data["trail_hi"] = close
                self._ld_log_msg(
                    f"[{sym}] 🟡 T1 ENTRY @ ${close:.2f}  RSI: {rsi:.1f}  "
                    f"Qty: {qty:.2f}", "entry")
                notify(f"Rush Algo's — {sym} T1", f"{sym} @ ${close:.2f}")

            elif xu_t2 and data["tier"] == 1:
                qty     = (data["equity"] * CONFIG.get("t2_size",2.0) * lev) / close
                new_avg = (data["pos"]*data["entry"] + qty*close) / (data["pos"]+qty)
                data["pos"]   += qty; data["entry"] = new_avg; data["tier"] = 2
                self._ld_log_msg(
                    f"[{sym}] 🔴 T2 ADD @ ${close:.2f}  Avg: ${new_avg:.2f}", "entry")

            elif xu_t3 and data["tier"] == 2:
                qty     = (data["equity"] * CONFIG.get("t3_size",2.0) * lev) / close
                new_avg = (data["pos"]*data["entry"] + qty*close) / (data["pos"]+qty)
                data["pos"]   += qty; data["entry"] = new_avg; data["tier"] = 3
                self._ld_log_msg(
                    f"[{sym}] 🟣 T3 ADD @ ${close:.2f}  Avg: ${new_avg:.2f}", "entry")

        # ── Exit signal ───────────────────────────────────────────
        if data["pos"] > 0:
            if xo_ex or (above_50 and data["tier"] > 0):
                pnl = (close - data["entry"]) * data["pos"]
                data["equity"] += pnl
                data["pos"]  = 0.0; data["entry"]  = 0.0
                data["tier"] = 0;   data["trail_hi"] = 0.0
                if pnl >= 0:
                    data["wins"] += 1; self._ld_session_wins += 1
                else:
                    data["losses"] += 1; self._ld_session_losses += 1
                self._ld_session_pnl += pnl
                pnl_str = f"+${pnl:.2f}" if pnl >= 0 else f"${pnl:.2f}"
                icon = "✅" if pnl >= 0 else "❌"
                self._ld_log_msg(
                    f"[{sym}] {icon} EXIT @ ${close:.2f}  P&L: {pnl_str}  "
                    f"Equity: ${data['equity']:.2f}", "exit")
                notify(f"Rush Algo's — {sym} EXIT",
                       f"{sym} exit @ ${close:.2f}  P&L: {pnl_str}")

        # ── Update card widgets ───────────────────────────────────
        try:
            rc = (S["purple"] if rsi and rsi < t3_rsi else
                  S["red"]    if rsi and rsi < t2_rsi else
                  S["orange"] if rsi and rsi < t1_rsi else
                  S["green"]  if rsi and rsi > ex_rsi  else S["text"])

            w["price"].config(text=f"${close:.2f}", fg=S["yellow"])
            # Populate company name once (when label is still empty)
            if w.get("name") and w["name"].cget("text") == "":

            w["rsi"].config(text=f"{rsi:.1f}" if rsi else "--", fg=rc if rsi else S["subtext"])
            w["sma200"].config(
                text="ABOVE ✓" if above_200 else "BELOW ✗",
                fg=S["green"] if above_200 else S["red"])

            live_pnl = (close - data["entry"]) * data["pos"] if data["pos"] > 0 else 0
            pnl_col  = S["green"] if live_pnl >= 0 else S["red"]
            pos_names = ["NONE","T1","T1+T2","FULL"]
            pos_cols  = [S["subtext"],S["orange"],S["red"],S["purple"]]
            tier = min(data["tier"], 3)

            w["pos"].config(text=pos_names[tier], fg=pos_cols[tier])
            w["entry"].config(
                text=f"${data['entry']:.2f}" if data["pos"] > 0 else "--",

            w["pnl"].config(
                text=f"{'+' if live_pnl>=0 else ''}${live_pnl:.2f}" if data["pos"] > 0 else "--",
                fg=pnl_col)
            w["equity"].config(text=f"${data['equity']:.2f}",
                               fg=S["green"] if data["equity"] >= CONFIG["initial_capital"] else S["red"])
            w["wins"].config(text=str(data["wins"]), fg=S["green"])
            w["losses"].config(text=str(data["losses"]), fg=S["red"])

            # Big signal label
            if data["tier"] == 0:
                if not above_200:
                    w["sig"].config(text="BLOCKED", fg=S["red"])
                    w["sub"].config(text="Price below SMA 200")
                elif not slope_ok:
                    w["sig"].config(text="BLOCKED", fg=S["orange"])

                elif not s50_slope_ok:
                    w["sig"].config(text="BLOCKED", fg=S["orange"])

                elif not cool_ok:

                else:
                    w["sig"].config(text=f"WATCHING T1", fg=S["blue"])
                    w["sub"].config(
                        text=f"RSI {rsi:.1f} → needs {t1_rsi}" if rsi else "Loading RSI...")
            elif data["tier"] == 1:
                w["sig"].config(text="IN T1", fg=S["orange"])
                w["sub"].config(
                    text=f"Entry ${data['entry']:.2f}  P&L: {'+' if live_pnl>=0 else ''}${live_pnl:.2f}",

            elif data["tier"] == 2:
                w["sig"].config(text="T1 + T2", fg=S["red"])

    def _tick(self):
        try:
            while True:
                kind, payload = self.data_queue.get_nowait()
                if kind == "data":
                    is_first = self.df is None
                    self.df  = payload
                    self._update_topbar(payload)
                    # Rerun backtest on first load, interval change, or ticker change
                    pending = getattr(self, "_pending_rerun", False)
                    if is_first or pending:
                        self._pending_rerun = False
                        if self.bt_result is not None and not is_first:
                            self._rerun_current_backtest()
                        else:
                            self._run_default_backtest()
                elif kind == "error":
                    self.lbl_statusbar.config(text=f"[ERR] {payload}")
        except queue.Empty:
            pass

        self._update_clock()
        if self.flash_count > 0:
            self._do_flash()
        self.after(1000, self._tick)

    def _update_clock(self):
        n = datetime.datetime.now()
        self.lbl_time.config(text=n.strftime("%H:%M:%S"))
        if is_market_open():
            self.lbl_market.config(text="● MARKET OPEN", fg=S["green"])
        else:
            self.lbl_market.config(text="● MARKET CLOSED", fg=S["red"])

    def _update_topbar(self, df):
        last    = df.iloc[-1]
        close   = float(last["Close"])
        rsi_val = float(last["rsi"]) if not pd.isna(last["rsi"]) else None
        now_str = datetime.datetime.now().strftime("%H:%M:%S")

        self.lbl_price.config(text=f"${close:.2f}")
        self.lbl_updated.config(text=f"Last update: {now_str}")

        if rsi_val is not None:
            rc = (S["purple"] if rsi_val < CONFIG["t3_rsi"] else
                  S["red"]    if rsi_val < CONFIG["t2_rsi"] else
                  S["orange"] if rsi_val < CONFIG["t1_rsi"] else
                  S["green"]  if rsi_val > CONFIG["exit_rsi"] else S["text"])
            self.lbl_rsi_top.config(text=f"{rsi_val:.1f}", fg=rc)

        if not hasattr(self, "_session_start") or self._session_start is None:
            self._session_start = close
        pnl = (close - self._session_start) / self._session_start * CONFIG["initial_capital"]
        pc  = S["green"] if pnl >= 0 else S["red"]
        self.lbl_session_pnl.config(
            text=f"Session P&L: {'+' if pnl>=0 else ''}${pnl:.2f}", fg=pc)

        self.lbl_statusbar.config(
            text=f"{CONFIG['ticker']}  ${close:.2f}  |  "
                 f"RSI {rsi_val:.1f}  |  Bars: {len(df)}  |  "
                 f"Interval: {CONFIG['interval']}  |  Leverage: {CONFIG['leverage']}x")

    def _run_default_backtest(self):
        """Auto-run backtest on last N days when data first loads."""
        if self.df is None: return
        end   = self.df.index[-1].date()
        start = end - datetime.timedelta(days=CONFIG["default_bt_days"])
        start = max(start, self.df.index[0].date())
        self._execute_backtest(start, end)

    def _execute_backtest(self, start_date, end_date):
        """Filter data, run correct engine for active strategy, update UI."""
        if self.df is None: return

        mask   = (self.df.index.date >= start_date) & (self.df.index.date <= end_date)
        df_rng = self.df[mask].copy()

        if len(df_rng) < 20:
            messagebox.showwarning("Insufficient Data",
                f"Only {len(df_rng)} bars in range. Try a wider window.")
            return

        # Route to correct backtest engine
        if ACTIVE_STRATEGY == "Futures":
            equity_curve, stop_curve, trades = run_futures_backtest(df_rng, CONFIG)
        else:
            equity_curve, stop_curve, trades = run_backtest(df_rng, CONFIG)

        metrics = compute_metrics(equity_curve, trades, CONFIG["initial_capital"])

        # ── Buy & Hold benchmark ──────────────────────────────────
        bh_start  = float(df_rng["Close"].iloc[0])
        bh_end    = float(df_rng["Close"].iloc[-1])
        bh_return = (bh_end - bh_start) / bh_start * 100
        bh_equity = CONFIG["initial_capital"] * (1 + bh_return / 100)
        # Equity curve for B&H — scales with price
        bh_curve  = (df_rng["Close"].values / bh_start) * CONFIG["initial_capital"]
        metrics["bh_return"] = round(bh_return, 2)
        metrics["bh_equity"]   = round(bh_equity, 2)
        metrics["bh_curve"]    = bh_curve
        metrics["alpha"]       = round(metrics["total_return"] - bh_return, 2)
        metrics["trades_raw"]  = trades
        metrics["df"]          = df_rng
        metrics["start"]       = start_date
        metrics["end"]         = end_date
        metrics["equity_curve"]= equity_curve
        metrics["stop_curve"]  = stop_curve
        self.bt_result         = metrics

        self._draw_backtest_chart(df_rng, equity_curve, stop_curve, trades, metrics)
        self._update_stats_panel(metrics)
        self._populate_trade_list(trades)

        fmt = "%b %d %Y"
        self.lbl_date_range.config(
            text=f"{start_date.strftime(fmt)}  →  {end_date.strftime(fmt)}  "
                 f"[{ACTIVE_STRATEGY}]")
        self.notebook.select(0)

        # Show comparison popup if this was triggered by a settings change
        if getattr(self, "_settings_changed_rerun", False) and \
           getattr(self, "_prev_bt_result", None):
            self._settings_changed_rerun = False
            self.after(200, lambda: self._show_settings_comparison(
                self._prev_bt_result, metrics,
                self._prev_config, dict(CONFIG)))

    # ─────────────────────────────────────────────────────────────
    # BACKTEST CHART DRAWING
    # ─────────────────────────────────────────────────────────────

    def _draw_backtest_chart(self, df, equity_curve, stop_curve, trades, metrics):
        self.ax_bt_price.cla()
        self.ax_bt_rsi.cla()

        for ax in [self.ax_bt_price, self.ax_bt_rsi]:
            ax.set_facecolor(S["panel"])
            ax.tick_params(colors=S["subtext"], labelsize=8)
            for sp in ax.spines.values():
                sp.set_edgecolor(S["border"])

        dates  = df.index
        closes = df["Close"].values
        opens  = df["Open"].values  if "Open"  in df.columns else closes
        highs  = df["High"].values  if "High"  in df.columns else closes
        lows   = df["Low"].values   if "Low"   in df.columns else closes
        rsi_v  = df["rsi"].values
        sma200 = df["sma200"].values
        sma50  = df["sma50"].values

        # ── Plot by bar INDEX not real time ───────────────────────
        # This removes all overnight/weekend gaps so candles are
        # evenly spaced exactly like TradingView displays them.
        x  = np.arange(len(df))
        cw = 0.6

        # ── Candlesticks ──────────────────────────────────────────
        for i in range(len(df)):
            o, h, l, c = opens[i], highs[i], lows[i], closes[i]
            if any(np.isnan([o,h,l,c])): continue
            bull = c >= o
            col  = S["green"] if bull else S["red"]
            self.ax_bt_price.plot([i, i], [l, h],
                                   color=col, lw=0.8, zorder=2)
            rect = mpatches.Rectangle(
                (i - cw/2, min(o,c)),
                cw, max(abs(c-o), cw*0.1),
                facecolor=col, edgecolor=col, lw=0, zorder=3, alpha=0.9)
            self.ax_bt_price.add_patch(rect)

        # ── SMAs ──────────────────────────────────────────────────
        self.ax_bt_price.plot(x, sma200, color=S["blue"],
                               lw=1.3, alpha=0.85, label="SMA 200", zorder=4)
        self.ax_bt_price.plot(x, sma50,  color=S["orange"],
                               lw=1.0, alpha=0.85, label="SMA 50",  zorder=4)

        # ── Stop levels ───────────────────────────────────────────
        stop_arr = np.array(stop_curve)
        valid    = ~np.isnan(stop_arr)
        if valid.any():
            self.ax_bt_price.plot(x[valid], stop_arr[valid],
                                   color=S["red"], lw=0.8,
                                   linestyle="--", alpha=0.6,
                                   label="Stop", zorder=4)

        # ── Buy & Hold reference line ─────────────────────────────
        if self.bt_result and "bh_curve" in self.bt_result:
            bh = self.bt_result["bh_curve"]
            if len(bh) == len(df):
                self.ax_bt_price.plot(x, bh, color=S["orange"],
                                       lw=0.8, linestyle=":",
                                       alpha=0.5, label="B&H", zorder=3)

        # ── Trade markers ─────────────────────────────────────────
        mkr = {"T1":  ("^",S["orange"],70,"T1 Entry"),
                "T2":  ("^",S["red"],   90,"T2 Add"),
                "T3":  ("^",S["purple"],110,"T3 Add"),
                "EXIT":("v",S["green"], 70,"Exit"),
                "STOP":("x",S["red"],   80,"Stop Hit")}

        buckets = {k: {"x":[],"y":[]} for k in mkr}
        for t in trades:
            k = t["type"]
            # T1_ENTRY stored by engine — map to T1 bucket for display
            if k == "T1_ENTRY": k = "T1"
            if k not in mkr: continue
            i = t["bar"]
            if i >= len(df): continue
            px  = t.get("price", t.get("exit", closes[i]))
            off = -0.004 if k in ("T1","T2","T3") else 0.004
            buckets[k]["x"].append(i)
            buckets[k]["y"].append(float(px) * (1 + off))

        for k, data in buckets.items():
            if not data["x"]: continue
            m, col, sz, lbl = mkr[k]
            self.ax_bt_price.scatter(data["x"], data["y"],
                                      marker=m, color=col, s=sz,
                                      zorder=6, label=lbl, edgecolors="none")

        # ── Chart formatting ──────────────────────────────────────
        pad = (np.nanmax(highs) - np.nanmin(lows)) * 0.04
        self.ax_bt_price.set_ylim(np.nanmin(lows)-pad, np.nanmax(highs)+pad)
        self.ax_bt_price.set_xlim(-1, len(df))
        self.ax_bt_price.set_ylabel("Price", color=S["subtext"], fontsize=8)
        ret_col = S["green"] if metrics["total_return"] >= 0 else S["red"]
        self.ax_bt_price.set_title(
            f"{CONFIG['ticker']}  |  {CONFIG['interval']}  |  "
            f"{metrics['start'].strftime('%b %d %Y')} -> {metrics['end'].strftime('%b %d %Y')}  |  "
            f"Return: {metrics['total_return']:+.2f}%  |  "
            f"{metrics['total_trades']} trades",
            color=ret_col, fontsize=9, pad=6)
        self.ax_bt_price.legend(loc="upper left", fontsize=6,
                                 facecolor=S["bg"], labelcolor=S["text"],
                                 framealpha=0.85, edgecolor=S["border"], ncol=3)

        # ── RSI panel ─────────────────────────────────────────────
        self.ax_bt_rsi.plot(x, rsi_v, color=S["purple"], lw=0.9)
        for lvl, col in [(CONFIG["t1_rsi"],S["orange"]),(CONFIG["t2_rsi"],S["red"]),
                          (CONFIG["t3_rsi"],S["purple"]),(CONFIG["exit_rsi"],S["green"])]:
            self.ax_bt_rsi.axhline(lvl, color=col, lw=0.5, linestyle="--", alpha=0.7)
        self.ax_bt_rsi.fill_between(x, 0, rsi_v,
                                     where=rsi_v < CONFIG["t3_rsi"],
                                     color=S["purple"], alpha=0.2)
        self.ax_bt_rsi.set_ylim(0, 100)
        self.ax_bt_rsi.set_xlim(-1, len(df))
        self.ax_bt_rsi.set_ylabel("RSI", color=S["subtext"], fontsize=8)

        # ── X-AXIS: date labels mapped from bar index ─────────────
        # Pick ~10 evenly spaced ticks and label with real dates
        n_ticks   = min(10, len(df))
        tick_locs = np.linspace(0, len(df)-1, n_ticks, dtype=int)
        tick_lbls = []
        for i in tick_locs:
            try:
                d = dates[i]
                tick_lbls.append(d.strftime("%b %d") if hasattr(d,"strftime")
                                  else str(d)[:10])
            except Exception:
                tick_lbls.append("")

        self.ax_bt_rsi.set_xticks(tick_locs)
        self.ax_bt_rsi.set_xticklabels(tick_lbls, fontsize=9,
                                        color=S["text"], rotation=0)
        self.ax_bt_price.set_xticks([])

        # Grid
        for ax in [self.ax_bt_price, self.ax_bt_rsi]:
            for loc in tick_locs:
                ax.axvline(loc, color=S["border"], lw=0.4, alpha=0.5)
            ax.yaxis.grid(True, color=S["border"], lw=0.3, alpha=0.4)

        self.canvas_bt.draw_idle()

    # STATS PANEL UPDATE

    def _update_stats_panel(self, metrics):
        fmt = "%b %d %Y"
        self.lbl_bt_period.config(
            text=f"{metrics['start'].strftime(fmt)}  →  {metrics['end'].strftime(fmt)}",
            fg=S["blue"])

        formats = {
            "total_return":  lambda v: (f"{v:+.2f}%",  S["green"] if v>=0 else S["red"]),
            "final_equity":  lambda v: (f"${v:,.2f}",  S["green"] if v>=CONFIG["initial_capital"] else S["red"]),
            "max_drawdown":  lambda v: (f"{v:.2f}%",   S["red"]),
            "sharpe":        lambda v: (f"{v:.3f}",     S["green"] if v>=1 else S["orange"] if v>=0 else S["red"]),
            "win_rate":      lambda v: (f"{v:.1f}%",   S["green"] if v>=50 else S["red"]),
            "profit_factor": lambda v: (f"{v:.3f}" if v!=float("inf") else "∞",
                                        S["green"] if v>=1 else S["red"]),
            "avg_win":       lambda v: (f"${v:+.2f}",  S["green"]),
            "avg_loss":      lambda v: (f"${v:+.2f}",  S["red"]),
            "total_trades":  lambda v: (str(v),         S["text"]),
        }
        for key, (lbl, unit) in self.stat_labels.items():
            if key in metrics:
                text, color = formats[key](metrics[key])
                lbl.config(text=text, fg=color)

        # ── Benchmark labels ──────────────────────────────────────
        bh_ret    = metrics.get("bh_return", 0)
        bh_eq     = metrics.get("bh_equity", CONFIG["initial_capital"])
        alpha     = metrics.get("alpha", 0)
        outperforms = alpha > 0

        self.bmark_labels["bh_return"].config(
            text=f"{bh_ret:+.2f}%",
            fg=S["green"] if bh_ret >= 0 else S["red"])
        self.bmark_labels["bh_equity"].config(
            text=f"${bh_eq:,.2f}",
            fg=S["green"] if bh_eq >= CONFIG["initial_capital"] else S["red"])
        self.bmark_labels["alpha"].config(
            text=f"{alpha:+.2f}%",
            fg=S["green"] if alpha > 0 else S["red"])
        self.bmark_labels["outperforms"].config(
            text="YES ✓" if outperforms else "NO ✗",
            fg=S["green"] if outperforms else S["red"])

        # Equity curve mini chart — strategy vs B&H
        self.ax_eq.cla()
        self.ax_eq.set_facecolor(S["panel"])
        eq   = metrics["equity_curve"]
        xeq  = np.arange(len(eq))
        col  = S["green"] if eq[-1] >= CONFIG["initial_capital"] else S["red"]

        # Strategy line
        self.ax_eq.plot(eq, color=col, lw=1.4, label="Strategy", zorder=3)
        self.ax_eq.fill_between(xeq, CONFIG["initial_capital"], eq,
                                 color=col, alpha=0.12)

        # Buy & Hold line — resample to match equity curve length
        bh_curve = metrics.get("bh_curve", None)
        if bh_curve is not None and len(bh_curve) > 0:
            # Resample B&H to same length as equity curve
            bh_x_orig = np.linspace(0, len(eq)-1, len(bh_curve))
            bh_interp  = np.interp(xeq, bh_x_orig, bh_curve)
            self.ax_eq.plot(bh_interp, color=S["orange"], lw=1.0,
                            linestyle="--", alpha=0.85,
                            label=f"B&H {bh_ret:+.1f}%", zorder=2)

        self.ax_eq.axhline(CONFIG["initial_capital"], color=S["subtext"],
                            lw=0.5, linestyle=":", alpha=0.6)
        self.ax_eq.set_title(
            f"Equity  |  Strategy {metrics['total_return']:+.1f}%  vs  "
            f"B&H {bh_ret:+.1f}%",
            color=S["orange"] if outperforms else S["red"],
            fontsize=6.5, pad=3)
        self.ax_eq.legend(loc="upper left", fontsize=5.5,
                           facecolor=S["bg"], labelcolor=S["text"],
                           framealpha=0.8, edgecolor=S["border"])
        self.ax_eq.tick_params(colors=S["subtext"], labelsize=6)

        self.ax_dd.tick_params(colors=S["subtext"], labelsize=6)

            plt.FuncFormatter(lambda v,_: f"{v:.0f}%"))
        for sp in self.ax_dd.spines.values(): sp.set_edgecolor(S["border"])
        self.canvas_dd.draw_idle()

    # ─────────────────────────────────────────────────────────────
    # TRADE LIST POPULATION
    # ─────────────────────────────────────────────────────────────

    def _populate_trade_list(self, trades):
        # Clear existing rows — reset selection reference first to avoid
        # TclError when callbacks try to configure destroyed widgets
        self._selected_row_widgets = None
        for w in self.tl_frame.winfo_children():
            w.destroy()
        self._trade_counter = 0
        self._cum_pnl       = 0.0

        exits = [t for t in trades if t["type"] in ("EXIT","STOP")]
        for t in exits:
            self._add_trade_row(t)

        # Summary
        pnls    = [t["pnl"] for t in exits if "pnl" in t]
        wins    = [p for p in pnls if p > 0]
        losses  = [p for p in pnls if p <= 0]
        total   = len(pnls)
        wr      = len(wins)/total*100 if total else 0
        tot_pnl = sum(pnls)
        avg_win = sum(wins)/len(wins)   if wins   else 0
        avg_los = sum(losses)/len(losses) if losses else 0
        pf      = abs(sum(wins)/sum(losses)) if losses and sum(losses) != 0 else float("inf")

        self.tl_stats["tl_total"].config(text=str(total))
        self.tl_stats["tl_wr"].config(text=f"{wr:.1f}%",
                                       fg=S["green"] if wr>=50 else S["red"])
        self.tl_stats["tl_pnl"].config(text=f"${tot_pnl:+.2f}",
                                        fg=S["green"] if tot_pnl>=0 else S["red"])
        self.tl_stats["tl_avgwin"].config(text=f"${avg_win:.2f}")
        self.tl_stats["tl_avgloss"].config(text=f"${avg_los:.2f}")
        self.tl_stats["tl_pf"].config(
            text=f"{pf:.2f}" if pf!=float("inf") else "∞",
            fg=S["green"] if pf>=1 else S["red"])

    def _add_trade_row(self, trade):
        self._trade_counter += 1
        self._cum_pnl       += trade["pnl"]
        pnl      = trade["pnl"]
        entry_px = float(trade.get("entry", trade.get("price", 0)))
        qty      = float(trade.get("qty", 0))
        pnl_pct  = (pnl / (entry_px * qty) * 100) if entry_px * qty != 0 else 0
        is_win   = pnl >= 0
        pc       = S["green"] if is_win else S["red"]
        cum_col  = S["green"] if self._cum_pnl >= 0 else S["red"]
        row_bg   = S["panel"] if self._trade_counter % 2 == 0 else S["bg"]
        trade_num = self._trade_counter

        row = tk.Frame(self.tl_frame, bg=row_bg, cursor="hand2")
        row.pack(fill="x")
        tk.Frame(row, bg=pc, width=3).pack(side="left", fill="y")

        def fmt_date(d):
            if hasattr(d, "strftime"): return d.strftime("%Y-%m-%d %H:%M")
            return str(d) if d else "--"

        # Entry date = when T1 fired, Exit date = when trade closed
        entry_date_str = fmt_date(trade.get("entry_date") or trade.get("date", ""))
        exit_date_str  = fmt_date(trade.get("date", ""))

        vals = [
            (str(self._trade_counter),              S["subtext"], 4),
            (trade["type"],                          pc,           6),
            (entry_date_str,                         S["text"],    14),
            (f"${entry_px:.2f}",                     S["text"],    9),
            (exit_date_str,                          S["text"],    14),
            (f"${float(trade.get('exit', entry_px)):.2f}", S["text"], 9),
            # For futures show contracts, for stocks show shares
            (f"{qty:.0f} {'cts' if ACTIVE_STRATEGY=='Futures' else 'sh'}",
                                                     S["subtext"], 8),
            # Show points for futures trades
            (f"{trade['points']:+.2f}pts" if "points" in trade else "--",
                                                     pc,           9) if ACTIVE_STRATEGY=="Futures"
             else (f"{'+' if is_win else ''}{pnl:.2f}", pc, 9),
            (f"{'+' if is_win else ''}{pnl:.2f}",   pc,           10),
            (f"{'+' if is_win else ''}{pnl_pct:.1f}%", pc,        8),
            (f"{'+' if self._cum_pnl>=0 else ''}{self._cum_pnl:.2f}", cum_col, 11),
        ]

        children = []
        for val, col, width in vals:
            lbl = tk.Label(row, text=val, font=("Courier",8),
                           bg=row_bg, fg=col, width=width, anchor="w",
                           cursor="hand2")
            lbl.pack(side="left", padx=2, pady=2)
            children.append(lbl)

        # ── Click to zoom chart to this trade ─────────────────────
        def on_click(e, t=trade, bg=row_bg, r=row, c=children, n=trade_num):
            # Highlight selected row
            if hasattr(self, "_selected_row_widgets") and self._selected_row_widgets:
                prev_row, prev_children, prev_bg = self._selected_row_widgets
                prev_row.config(bg=prev_bg)
                for w in prev_children:
                    w.config(bg=prev_bg)
            r.config(bg="#1c2a3a")
            for w in c:
                w.config(bg="#1c2a3a")
            self._selected_row_widgets = (r, c, bg)
            self._zoom_to_trade(t, n)
            # Switch to backtest tab
            self.notebook.select(0)

        def on_enter(e, r=row, c=children):
            if not (hasattr(self, "_selected_row_widgets") and
                    self._selected_row_widgets and
                    self._selected_row_widgets[0] is r):
                r.config(bg="#1a2332")
                for w in c: w.config(bg="#1a2332")

        def on_leave(e, r=row, c=children, bg=row_bg):
            if not (hasattr(self, "_selected_row_widgets") and
                    self._selected_row_widgets and
                    self._selected_row_widgets[0] is r):
                r.config(bg=bg)
                for w in c: w.config(bg=bg)

        row.bind("<Button-1>", on_click)
        row.bind("<Enter>",    on_enter)
        row.bind("<Leave>",    on_leave)
        for w in children:
            w.bind("<Button-1>", on_click)
            w.bind("<Enter>",    on_enter)
            w.bind("<Leave>",    on_leave)

    # ─────────────────────────────────────────────────────────────
    # TRADE ZOOM
    # ─────────────────────────────────────────────────────────────

    def _zoom_to_trade(self, trade, trade_num):
        """Zoom the backtest chart to a specific trade with context candles."""
        if self.bt_result is None: return

        self._zoom_active = True
        self._zoom_trade  = (trade, trade_num)

        # Show zoom bar
        self._zoom_bar.pack(fill="x", before=self.canvas_bt.get_tk_widget())

        self._render_zoom(trade, trade_num)

    def _render_zoom(self, trade, trade_num):
        df     = self.bt_result.get("df")
        trades = self.bt_result.get("trades_raw", [])
        if df is None: return

        ctx      = self._zoom_context.get()
        exit_bar = trade.get("bar", 0)

        # Find true entry bar + all tiers for this specific trade
        entry_bar   = exit_bar
        entry_tiers = []
        for t in reversed(trades):
            b = t.get("bar", 0)
            if b >= exit_bar: continue
            if t["type"] in ("T1","T1_ENTRY","GAP_UP_SHORT","GAP_DN_LONG"):
                entry_bar = b
                entry_tiers.insert(0, t)  # T1 goes first
                break
            elif t["type"] in ("T2","T3"):
                entry_tiers.insert(0, t)  # collect adds in order

        # Sort all tiers by bar index to guarantee chronological order
        entry_tiers.sort(key=lambda t: t.get("bar", 0))

        # Window: ctx before entry, ctx after exit
        start_bar = max(0, entry_bar - ctx)
        end_bar   = min(len(df) - 1, exit_bar + ctx)

        df_zoom = df.iloc[start_bar:end_bar+1].copy()
        df_zoom.reset_index(drop=False, inplace=True)

        pnl      = trade.get("pnl", 0)
        is_win   = pnl >= 0
        pnl_col  = S["green"] if is_win else S["red"]
        entry_px = float(trade.get("entry", trade.get("price", 0)))
        exit_px  = float(trade.get("exit", entry_px))
        result   = "WIN" if is_win else "LOSS"

        pnl_str0 = f"+${pnl:.2f}" if is_win else f"${pnl:.2f}"
        self._lbl_zoom_info.config(
            text=(f"Trade #{trade_num}  |  {trade['type']}  |  "
                  f"Entry ${entry_px:.2f}  ->  Exit ${exit_px:.2f}  |  "
                  f"P&L: {pnl_str0}  [{result}]"),
            fg=pnl_col)

        self.ax_bt_price.cla()
        self.ax_bt_rsi.cla()

        for ax in [self.ax_bt_price, self.ax_bt_rsi]:
            ax.set_facecolor(S["panel"])
            ax.tick_params(colors=S["subtext"], labelsize=8)
            for sp in ax.spines.values():
                sp.set_edgecolor(S["border"])

        orig_col = None
        for c in ["Datetime","Date","index"]:
            if c in df_zoom.columns:
                orig_col = c; break
        dt_series = pd.to_datetime(df_zoom[orig_col]) if orig_col else pd.to_datetime(df_zoom.index)

        # Use bar index for x — no gaps
        x_zoom = np.arange(len(df_zoom))
        cw     = 0.6

        closes = df_zoom["Close"].values
        opens  = df_zoom["Open"].values  if "Open"  in df_zoom.columns else closes
        highs  = df_zoom["High"].values  if "High"  in df_zoom.columns else closes
        lows   = df_zoom["Low"].values   if "Low"   in df_zoom.columns else closes
        rsi_v  = df_zoom["rsi"].values   if "rsi"   in df_zoom.columns else np.full(len(closes),50)
        sma200 = df_zoom["sma200"].values if "sma200" in df_zoom.columns else None
        sma50  = df_zoom["sma50"].values  if "sma50"  in df_zoom.columns else None

        entry_zi = max(0, entry_bar - start_bar)
        exit_zi  = min(len(x_zoom)-1, exit_bar - start_bar)
        cw = (x_zoom[-1]-x_zoom[0])/len(x_zoom)*0.6 if len(x_zoom)>1 else 0.01

        # Candles — dim outside trade window
        for i in range(len(df_zoom)):
            o,h,l,c = opens[i],highs[i],lows[i],closes[i]
            if any(np.isnan([o,h,l,c])): continue
            in_trade = entry_zi <= i <= exit_zi
            if in_trade:
                col   = S["green"] if c>=o else S["red"]
                alpha = 0.95
            else:
                col   = S["subtext"]
                alpha = 0.25
            self.ax_bt_price.plot([x_zoom[i],x_zoom[i]],[l,h],
                                   color=col,lw=0.8,alpha=alpha,zorder=2)
            rect = mpatches.Rectangle(
                (x_zoom[i]-cw/2,min(o,c)),cw,max(abs(c-o),cw*0.1),
                facecolor=col,edgecolor=col,lw=0,zorder=3,alpha=alpha)
            self.ax_bt_price.add_patch(rect)

        if sma200 is not None:
            self.ax_bt_price.plot(x_zoom,sma200,color=S["blue"],lw=1.2,alpha=0.8,label="SMA 200")
        if sma50 is not None:
            self.ax_bt_price.plot(x_zoom,sma50,color=S["orange"],lw=1.0,alpha=0.8,label="SMA 50")

        # THIS TRADE ONLY — entry markers
        if 0 <= entry_zi < len(x_zoom):
            self.ax_bt_price.scatter([x_zoom[entry_zi]],[entry_px*0.996],
                marker="^",color=S["orange"],s=160,zorder=8,edgecolors=S["bg"],linewidths=0.8)
            self.ax_bt_price.annotate(
                f"T1 ENTRY\n${entry_px:.2f}",
                (x_zoom[entry_zi],entry_px*0.992),
                color=S["orange"],fontsize=8,ha="center",va="top",fontweight="bold")

        for t in entry_tiers:
            if t["type"] in ("T2","T3"):
                tz = t["bar"]-start_bar
                if 0 <= tz < len(x_zoom):
                    tc  = S["red"] if t["type"]=="T2" else S["purple"]
                    tpx = float(t.get("price",closes[tz]))
                    ttype = t["type"]
                    self.ax_bt_price.scatter([x_zoom[tz]],[tpx*0.996],
                        marker="^",color=tc,s=180,zorder=8,edgecolors=S["bg"],linewidths=0.8)
                    self.ax_bt_price.annotate(
                        f"{ttype} ADD\n${tpx:.2f}",
                        (x_zoom[tz],tpx*0.992),
                        color=tc,fontsize=8,ha="center",va="top",fontweight="bold")

        if 0 <= exit_zi < len(x_zoom):
            if trade["type"]=="STOP":
                self.ax_bt_price.scatter([x_zoom[exit_zi]],[exit_px*1.005],

        for zi, col in [(entry_zi,S["orange"]),(exit_zi,pnl_col)]:

        self.ax_bt_price.legend(loc="upper left",fontsize=6,

        pnl_str2  = f"+${pnl:.2f}" if is_win else f"${pnl:.2f}"
        ttype_str = trade["type"]
        self.ax_bt_price.set_title(
            f"Trade #{trade_num}  -  {ttype_str}  -  "

            f"P&L: {pnl_str2}  [{result}]",
            color=pnl_col,fontsize=9,pad=6)

        if 0<entry_zi<=len(rsi_v) and 0<exit_zi<=len(rsi_v):
            if entry_zi>0:
                self.ax_bt_rsi.plot(x_zoom[:entry_zi+1],rsi_v[:entry_zi+1],
                    color=S["subtext"],lw=0.7,alpha=0.3)
            self.ax_bt_rsi.plot(x_zoom[entry_zi:exit_zi+1],rsi_v[entry_zi:exit_zi+1],
                color=S["purple"],lw=1.3)
            if exit_zi<len(rsi_v)-1:
                self.ax_bt_rsi.plot(x_zoom[exit_zi:],rsi_v[exit_zi:],
                    color=S["subtext"],lw=0.7,alpha=0.3)
        else:
            self.ax_bt_rsi.plot(x_zoom,rsi_v,color=S["purple"],lw=0.9)

        for lvl,col in [(CONFIG["t1_rsi"],S["orange"]),(CONFIG["t2_rsi"],S["red"]),

            self.ax_bt_rsi.axhline(lvl,color=col,lw=0.5,linestyle="--",alpha=0.7)

        self.ax_bt_rsi.set_ylabel("RSI",color=S["subtext"],fontsize=8)

        # X axis — index based, labeled with real dates
        n_ticks   = min(10, len(df_zoom))
        tick_locs = np.linspace(0, len(df_zoom)-1, n_ticks, dtype=int)
        tick_lbls = []
        for ti in tick_locs:
            try:
                d = dt_series.iloc[ti]
                tick_lbls.append(d.strftime("%b %d %H:%M") if hasattr(d,"strftime") else str(d))
            except Exception:
                tick_lbls.append("")
        self.ax_bt_rsi.set_xticks(tick_locs)
        self.ax_bt_rsi.set_xticklabels(tick_lbls, fontsize=7.5, color=S["text"], rotation=0)
        self.ax_bt_price.set_xticks([])
        for ax in [self.ax_bt_price,self.ax_bt_rsi]:

                ax.axvline(loc,color=S["border"],lw=0.4,alpha=0.5)
            ax.yaxis.grid(True,color=S["border"],lw=0.3,alpha=0.4)
        self.canvas_bt.draw_idle()

    def _run_wfo(self):
        """Launch walk-forward in a background thread."""
        if self.df is None:
            messagebox.showwarning("No Data", "Load data first by running a backtest.")
            return

        if str(CONFIG.get("wfo_enabled","false")).lower() != "true":
            if not messagebox.askyesno("Walk-Forward Disabled",
                "Walk-Forward is currently disabled in Settings.\n\n"
                "Run anyway?"):
                return

        # ── Time range picker popup ───────────────────────────────
        popup = tk.Toplevel(self)
        popup.title("Walk-Forward — Select Time Range")
        popup.configure(bg=S["bg"])
        popup.resizable(False, False)
        popup.geometry(f"460x340+{self.winfo_x()+200}+{self.winfo_y()+150}")
        popup.grab_set()

        tk.Label(popup, text="Select Walk-Forward Time Range",
                 font=("Courier",12,"bold"),
                 bg=S["bg"], fg=S["blue"]).pack(pady=(18,4))
        tk.Label(popup,
                 text="Optimize on recent data for current regime.\n"
                      "Older windows show long-term robustness.",
                 font=("Courier",8), bg=S["bg"],
                 fg=S["subtext"], justify="center").pack(pady=(0,12))

        data_start  = self.df.index[0].date()
        data_end    = self.df.index[-1].date()
        total_days  = (data_end - data_start).days
        bars_per_day = len(self.df) / max(total_days, 1)

        presets = []
        for label, days in [("3 Months",      90),
                             ("6 Months",      180),
                             ("1 Year",        365),
                             ("2 Years",       730),
                             ("All Available", total_days)]:
            if days <= total_days + 7:
                actual_start = max(data_end - datetime.timedelta(days=days), data_start)
                actual_days  = (data_end - actual_start).days
                est_bars     = int(actual_days * bars_per_day)
                presets.append((label, days, actual_start, est_bars))

        selected = tk.StringVar(value=presets[-1][0])

        btn_frame = tk.Frame(popup, bg=S["bg"])
        btn_frame.pack(pady=2, fill="x", padx=30)

        for label, days, actual_start, est_bars in presets:
            row = tk.Frame(btn_frame, bg=S["bg"])
            row.pack(fill="x", pady=2)
            tk.Radiobutton(
                row, text=f"  {label:<14}",
                variable=selected, value=label,
                font=("Courier",10,"bold"), bg=S["bg"],
                fg=S["text"], selectcolor=S["panel"],
                activebackground=S["bg"],
                relief="flat").pack(side="left")
            tk.Label(row,
                     text=f"{actual_start.strftime('%b %d %Y')} → {data_end.strftime('%b %d %Y')}"
                          f"  ({est_bars:,} bars)",
                     font=("Courier",8), bg=S["bg"],
                     fg=S["subtext"]).pack(side="left", padx=(4,0))

        tk.Frame(popup, bg=S["border"], height=1).pack(fill="x", padx=20, pady=(12,0))

        def on_run():
            lookup = {lbl: (d, s) for lbl, d, s, _ in presets}
            days, start = lookup[selected.get()]
            popup.destroy()
            self._launch_wfo(start, data_end)

        bf = tk.Frame(popup, bg=S["bg"])
        bf.pack(pady=14)
        tk.Button(bf, text="  ▶  Run Walk-Forward  ",
                  font=("Courier",10,"bold"),
                  bg=S["green"], fg=S["bg"],
                  relief="flat", cursor="hand2",
                  command=on_run).pack(side="left", padx=8)
        tk.Button(bf, text="  Cancel  ",
                  font=("Courier",9),
                  bg=S["panel"], fg=S["subtext"],
                  relief="flat", cursor="hand2",
                  command=popup.destroy).pack(side="left", padx=4)

    def _launch_wfo(self, start_date, end_date):
        """Start WFO thread on the selected date range."""
        if self.df is None: return

        mask   = (self.df.index.date >= start_date) & \
                 (self.df.index.date <= end_date)
        df_wfo = self.df[mask].copy()

        if len(df_wfo) < 100:
            messagebox.showwarning("Insufficient Data",
                f"Only {len(df_wfo)} bars in selected range. Choose a wider window.")
            return

        self._wfo_run_btn.config(text="  ⏳  Running...  ",
                                  bg=S["orange"], state="disabled")
        self._wfo_prog_lbl.config(
            text=f"Starting — {start_date.strftime('%b %d %Y')} → "
                 f"{end_date.strftime('%b %d %Y')}  ({len(df_wfo)} bars)",
            fg=S["yellow"])

        def worker():
            def progress(pct, msg):
                self.after(0, lambda: self._wfo_update_progress(pct, msg))
            result, err = run_walk_forward(df_wfo, CONFIG, progress_cb=progress)
            self.after(0, lambda: self._wfo_show_results(result, err))

        threading.Thread(target=worker, daemon=True).start()

    def _wfo_update_progress(self, pct, msg):
        """Update progress bar and label from main thread."""
        self._wfo_prog_lbl.config(text=msg, fg=S["yellow"])
        w = self._wfo_prog_bar.winfo_width()
        if w < 2: return
        self._wfo_prog_bar.delete("all")
        fill_w = int(w * pct / 100)
        col = S["green"] if pct >= 100 else S["blue"]
        self._wfo_prog_bar.create_rectangle(
            0, 0, fill_w, 14, fill=col, outline="")
        self._wfo_prog_bar.create_text(
            w//2, 7, text=f"{pct}%",
            fill=S["text"], font=("Courier",7))

    def _wfo_show_results(self, result, err):
        """Populate fold table and chart with WFO results."""
        self._wfo_run_btn.config(text="  ▶  Run Walk-Forward  ",
                                  bg=S["green"], state="normal")

        if err or result is None:
            self._wfo_prog_lbl.config(
                text=f"Error: {err}", fg=S["red"])
            return

        self._wfo_prog_lbl.config(
            text=f"Complete — {len(result['fold_results'])} folds",
            fg=S["green"])

        # Clear old rows
        for w in self._wfo_fold_frame.winfo_children():
            w.destroy()

        # Fill fold table
        for fr in result["fold_results"]:
            is_win = fr["oos_return"] >= 0
            rc     = S["green"] if is_win else S["red"]
            row    = tk.Frame(self._wfo_fold_frame,
                              bg=S["panel"] if fr["fold"] % 2 else S["bg"])
            row.pack(fill="x")

            def cell(text, fg=S["text"], width=6):
                tk.Label(row, text=text, font=("Courier",7),
                         bg=row["bg"], fg=fg,
                         width=width, anchor="w").pack(side="left", padx=2, pady=2)

            cell(str(fr["fold"]), width=4)
            ts = fr["train_start"].strftime("%m/%d/%y") if hasattr(fr["train_start"],"strftime") else str(fr["train_start"])[:8]
            te = fr["train_end"].strftime("%m/%d/%y")   if hasattr(fr["train_end"],  "strftime") else str(fr["train_end"])[:8]
            os_s = fr["test_start"].strftime("%m/%d/%y") if hasattr(fr["test_start"],"strftime") else str(fr["test_start"])[:8]
            os_e = fr["test_end"].strftime("%m/%d/%y")   if hasattr(fr["test_end"],  "strftime") else str(fr["test_end"])[:8]
            cell(f"{ts}→{te}", width=20)
            cell(f"{os_s}→{os_e}", width=20)
            cell(str(fr["best_t1"]),  width=4)
            cell(str(fr["best_exit"]), width=4)
            cell(f"{fr['best_stop']:.1f}", width=5)
            cell(f"{fr['oos_return']:+.1f}%", fg=rc, width=8)
            cell(f"{fr['oos_sharpe']:.2f}",   fg=rc, width=9)
            cell(f"{fr['oos_win_rate']:.0f}%", width=6)
            cell(str(fr["oos_trades"]), width=6)

        # ── Consistency analysis ──────────────────────────────────
        folds = result["fold_results"]

        # Find modal (most common) best params across folds
        from collections import Counter
        t1_counts   = Counter(fr["best_t1"]   for fr in folds)
        exit_counts = Counter(fr["best_exit"] for fr in folds)
        stop_counts = Counter(fr["best_stop"] for fr in folds)

        modal_t1   = t1_counts.most_common(1)[0]
        modal_exit = exit_counts.most_common(1)[0]
        modal_stop = stop_counts.most_common(1)[0]

        # Overfitting ratio: avg(OOS metric) / avg(train metric)
        train_metrics = [fr["train_metric"] for fr in folds if fr["train_metric"] > -900]
        oos_sharpes   = [fr["oos_sharpe"]   for fr in folds]
        avg_train = np.mean(train_metrics) if train_metrics else 0
        avg_oos   = np.mean(oos_sharpes)
        overfit_ratio = avg_oos / avg_train if avg_train > 0 else 0

        # Win rate across folds
        n_winning = sum(1 for fr in folds if fr["oos_return"] >= 0)
        fold_win_pct = n_winning / len(folds) * 100

        # Param stability score — how often did the same params win?
        stability = (modal_t1[1] + modal_exit[1] + modal_stop[1]) / (3 * len(folds)) * 100

        # Score interpretation
        if overfit_ratio >= 0.7 and fold_win_pct >= 60:
            verdict = "✓ ROBUST — edge appears real across folds"
            vc = S["green"]
        elif overfit_ratio >= 0.4 and fold_win_pct >= 40:
            verdict = "~ MODERATE — some consistency, monitor live"
            vc = S["yellow"]
        else:
            verdict = "✗ FRAGILE — likely overfit, don't deploy"
            vc = S["red"]

        consistency_text = (
            f"Modal params:  T1={modal_t1[0]} ({modal_t1[1]}/{len(folds)} folds)  "
            f"Exit={modal_exit[0]} ({modal_exit[1]}/{len(folds)})  "
            f"Stop={modal_stop[0]}% ({modal_stop[1]}/{len(folds)})  |  "
            f"Winning folds: {n_winning}/{len(folds)} ({fold_win_pct:.0f}%)  |  "
            f"OOS/Train Sharpe ratio: {overfit_ratio:.2f}  |  "
            f"Param stability: {stability:.0f}%"
        )
        self._wfo_consistency.config(text=consistency_text, fg=S["text"])

        # Verdict label — update summary to include verdict
        oos_ret = result["total_return"]
        col = S["green"] if oos_ret >= 0 else S["red"]
        self._wfo_summary.config(
            text=f"Combined OOS  |  Return: {oos_ret:+.2f}%  |  "
                 f"Sharpe: {result['sharpe']:.3f}  |  "
                 f"Win Rate: {result['win_rate']:.1f}%  |  "
                 f"Max DD: {result['max_drawdown']:.2f}%  |  "
                 f"Trades: {result['total_trades']}\n{verdict}",
            fg=vc)

        # Store best params for Apply button
        self._wfo_best_params = {
            "t1_rsi":      modal_t1[0],
            "t2_rsi":      max(modal_t1[0] - 8, 20),
            "t3_rsi":      max(modal_t1[0] - 15, 15),
            "exit_rsi":    modal_exit[0],
            "stop_pct_t1": modal_stop[0],
            "stop_pct_t2": max(modal_stop[0] - 1.0, 1.0),
            "stop_pct_t3": max(modal_stop[0] - 2.0, 1.0),
        }
        self._wfo_apply_btn.config(state="normal")

        # ── Draw OOS equity chart ─────────────────────────────────
        self._wfo_ax_eq.cla()
        self._wfo_ax_ret.cla()
        for ax in [self._wfo_ax_eq, self._wfo_ax_ret]:
            ax.set_facecolor(S["panel"])
            ax.tick_params(colors=S["subtext"], labelsize=7)
            for sp in ax.spines.values():
                sp.set_edgecolor(S["border"])

        eq  = np.array(result["oos_equity"])
        x   = np.arange(len(eq))
        cap = float(CONFIG["initial_capital"])

        # Shade folds alternating
        fold_size = int(CONFIG.get("wfo_test_bars", 250))
        for i, fr in enumerate(result["fold_results"]):
            x0 = i * fold_size
            x1 = min(x0 + fold_size, len(eq))
            col = "#1a2a1a" if i % 2 == 0 else "#1a1a2a"
            self._wfo_ax_eq.axvspan(x0, x1, color=col, alpha=0.4)

        # Equity line
        self._wfo_ax_eq.plot(x, eq, color=S["green"], lw=1.5,
                              label=f"OOS Equity  {oos_ret:+.1f}%")
        self._wfo_ax_eq.axhline(cap, color=S["subtext"],
                                 lw=0.6, linestyle="--", alpha=0.5,
                                 label=f"Capital ${cap:,.0f}")
        self._wfo_ax_eq.fill_between(x, cap, eq,
            where=eq >= cap, color=S["green"], alpha=0.08)
        self._wfo_ax_eq.fill_between(x, cap, eq,
            where=eq < cap,  color=S["red"],   alpha=0.08)
        self._wfo_ax_eq.set_ylabel("Equity ($)", color=S["subtext"], fontsize=7)
        self._wfo_ax_eq.legend(loc="upper left", fontsize=6,
                                facecolor=S["bg"], labelcolor=S["text"],
                                framealpha=0.8, edgecolor=S["border"])
        self._wfo_ax_eq.set_title(
            f"Walk-Forward OOS  |  {len(result['fold_results'])} folds  |  "
            f"Optimizing: {CONFIG.get('wfo_metric','sharpe').title()}",
            color=S["blue"], fontsize=8, pad=4)

        # Per-fold return bars
        fold_rets = [fr["oos_return"] for fr in result["fold_results"]]
        fold_x    = np.arange(len(fold_rets))
        fold_cols = [S["green"] if r >= 0 else S["red"] for r in fold_rets]
        self._wfo_ax_ret.bar(fold_x, fold_rets, color=fold_cols, alpha=0.8)
        self._wfo_ax_ret.axhline(0, color=S["subtext"], lw=0.5)
        self._wfo_ax_ret.set_ylabel("OOS Return %", color=S["subtext"], fontsize=7)
        self._wfo_ax_ret.set_xticks(fold_x)
        self._wfo_ax_ret.set_xticklabels(
            [f"F{fr['fold']}" for fr in result["fold_results"]],
            fontsize=7, color=S["text"])
        for ax in [self._wfo_ax_eq, self._wfo_ax_ret]:
            ax.yaxis.grid(True, color=S["border"], lw=0.3, alpha=0.4)

        self._wfo_canvas.draw_idle()

        # Switch to WFO tab
        self.notebook.select(3)

    def _wfo_apply_best_params(self):
        """Apply the modal best params from WFO to CONFIG and rerun backtest."""
        p = getattr(self, "_wfo_best_params", None)
        if not p:
            return

        # Show confirmation popup
        msg = "\n".join([
            "Apply these walk-forward optimized parameters?",
            "",
            f"  T1 Entry RSI:   {p['t1_rsi']}",
            f"  T2 Add RSI:     {p['t2_rsi']}",
            f"  T3 Add RSI:     {p['t3_rsi']}",
            f"  Exit RSI:       {p['exit_rsi']}",
            f"  T1 Stop %:      {p['stop_pct_t1']}",
            f"  T2 Stop %:      {p['stop_pct_t2']}",
            f"  T3 Stop %:      {p['stop_pct_t3']}",
            "",
            "This will update Settings and rerun the backtest.",
        ])
        if not messagebox.askyesno("Apply Walk-Forward Params", msg):
            return

        # Snapshot old for comparison popup
        self._prev_bt_result         = dict(self.bt_result) if self.bt_result else None
        self._prev_config            = dict(CONFIG)
        self._prev_strategy_settings = dict(
            STRATEGIES[ACTIVE_STRATEGY]["settings"])
        self._settings_changed_rerun = True

        # Apply
        CONFIG.update(p)
        STRATEGIES[ACTIVE_STRATEGY]["settings"].update(p)

        # Refresh settings form so fields reflect new values
        if hasattr(self, "_setting_vars"):
            for key, (var, kind) in self._setting_vars.items():
                if key in p:
                    var.set(str(p[key]))

        # Rerun backtest
        self._rerun_current_backtest()
        self.notebook.select(0)  # jump to backtest tab

    def _apply_settings(self):
        """Read setting vars, save to active strategy, apply to CONFIG, rerun."""
        global ACTIVE_STRATEGY
        errors   = []
        new_vals = {}

        for key, (var, kind) in self._setting_vars.items():
            raw = var.get().strip()
            if raw == "":
                continue  # skip fields not relevant to active strategy
            try:
                if kind == "choice":
                    new_vals[key] = str(raw)
                elif kind == "str":
                    new_vals[key] = str(raw)   # comma-separated ranges
                elif kind == "int":
                    new_vals[key] = int(float(raw))
                else:
                    new_vals[key] = float(raw)
            except ValueError:
                errors.append(f"{key}: '{raw}' is not a valid {kind}")

        if not errors:
            if new_vals.get("t1_rsi",40) <= new_vals.get("t2_rsi",32):
                errors.append("T1 RSI must be greater than T2 RSI")
            if new_vals.get("t2_rsi",32) <= new_vals.get("t3_rsi",25):
                errors.append("T2 RSI must be greater than T3 RSI")
            if new_vals.get("t3_rsi",25) >= new_vals.get("exit_rsi",50):
                errors.append("Exit RSI must be greater than T3 RSI")
            if new_vals.get("leverage",1.0) < 0.1:
                errors.append("Leverage must be at least 0.1")
            if new_vals.get("max_r2",1.0) < 0 or new_vals.get("max_r2",1.0) > 1:
                errors.append("R2 threshold must be between 0 and 1")

        if errors:
            self.lbl_settings_status.config(
                text="  " + "  |  ".join(errors), fg=S["red"])
            return

        interval_changed = new_vals.get("interval") != CONFIG.get("interval")

        # Snapshot old settings + result before applying new ones
        self._prev_bt_result  = dict(self.bt_result) if self.bt_result else None
        self._prev_config     = dict(CONFIG)
        self._prev_strategy_settings = {
            k: v for k, v in STRATEGIES[ACTIVE_STRATEGY]["settings"].items()
        }
        self._settings_changed_rerun = not interval_changed  # flag for comparison popup

        # Save into the active strategy's stored settings
        STRATEGIES[ACTIVE_STRATEGY]["settings"].update(new_vals)

        # Apply to live CONFIG
        CONFIG.update(new_vals)

        self.lbl_settings_status.config(
            text=f"Settings saved for '{ACTIVE_STRATEGY}'", fg=S["green"])
        self.update()

        if interval_changed:
            self.lbl_statusbar.config(text=f"Fetching {CONFIG['interval']} data...")
            # Update lookback_period to match the interval's max
            CONFIG["lookback_period"] = max_period_for_interval(CONFIG["interval"])
            STRATEGIES[ACTIVE_STRATEGY]["settings"]["lookback_period"] = CONFIG["lookback_period"]
            # Clear stale df so tick loop knows this is a fresh load
            self.df = None
            self._pending_rerun = True   # rerun backtest when new data arrives
            self.fetcher.stop()
            self.fetcher = DataFetcher(self.data_queue, CONFIG)
            self.fetcher.start()
            self.lbl_settings_status.config(
                text=f"Saved — fetching {CONFIG['interval']} data, backtest will rerun automatically...",
                fg=S["orange"])
        else:
            self._rerun_current_backtest()
            self.lbl_settings_status.config(
                text=f"'{ACTIVE_STRATEGY}' settings applied — backtest updated",
                fg=S["green"])

    def _show_settings_comparison(self, old_result, new_result, old_cfg, new_cfg):
        """Show side-by-side before/after popup after a settings change rerun."""
        popup = tk.Toplevel(self)
        popup.title("Settings Change — Results Comparison")
        popup.configure(bg=S["bg"])
        popup.resizable(False, False)
        popup.grab_set()  # modal

        # Center on main window
        popup.geometry(f"680x520+{self.winfo_x()+120}+{self.winfo_y()+80}")

        F_HDR  = ("Arial", 11, "bold")
        F_LBL  = ("Arial", 9)
        F_VAL  = ("Courier", 10, "bold")
        F_SEC  = ("Arial", 8, "bold")

        # ── Title ─────────────────────────────────────────────────
        tk.Label(popup, text="Settings Change — Impact on Backtest",
                 font=("Arial",13,"bold"),
                 bg=S["bg"], fg=S["blue"]).pack(pady=(16,2))
        tk.Label(popup, text="Review the changes below. Accept to keep new settings or Undo to revert.",
                 font=("Arial",8), bg=S["bg"], fg=S["subtext"]).pack(pady=(0,10))

        # ── Main body: three columns ───────────────────────────────
        body = tk.Frame(popup, bg=S["bg"])
        body.pack(fill="both", expand=True, padx=20)

        # Column headers
        for col, text, fg in [(0,"METRIC",S["subtext"]),
                               (1,"BEFORE",S["orange"]),
                               (2,"AFTER", S["blue"])]:
            tk.Label(body, text=text, font=F_HDR,
                     bg=S["bg"], fg=fg,
                     width=18, anchor="w" if col==0 else "center"
                     ).grid(row=0, column=col, padx=8, pady=(0,4))

        tk.Frame(body, bg=S["border"], height=1).grid(
            row=1, column=0, columnspan=3, sticky="ew", pady=(0,6))

        def metric_row(r, label, old_val, new_val, higher_better=True, is_pct=False):
            suffix = "%" if is_pct else ""
            try:
                o = float(old_val); n = float(new_val)
                improved = (n > o) if higher_better else (n < o)
                same     = abs(n - o) < 0.001
                n_col    = S["text"] if same else (S["green"] if improved else S["red"])
                delta    = n - o
                delta_str = f"  ({'+' if delta>=0 else ''}{delta:.2f}{suffix})"
            except:
                n_col = S["text"]; delta_str = ""

            tk.Label(body, text=label, font=F_LBL,
                     bg=S["bg"], fg=S["subtext"],
                     width=20, anchor="w").grid(row=r, column=0, padx=8, pady=3)
            tk.Label(body, text=f"{old_val}{suffix}", font=F_VAL,
                     bg=S["bg"], fg=S["orange"],
                     width=14, anchor="center").grid(row=r, column=1, padx=8)
            tk.Label(body, text=f"{new_val}{suffix}{delta_str}", font=F_VAL,
                     bg=S["bg"], fg=n_col,
                     width=22, anchor="center").grid(row=r, column=2, padx=8)

        # Metrics comparison
        metrics_to_show = [
            ("Total Return %",   "total_return",  True,  True),
            ("Final Equity $",   "final_equity",  True,  False),
            ("Max Drawdown %",   "max_drawdown",  False, True),
            ("Sharpe Ratio",     "sharpe",        True,  False),
            ("Win Rate %",       "win_rate",      True,  True),
            ("Profit Factor",    "profit_factor", True,  False),
            ("Avg Win $",        "avg_win",       True,  False),
            ("Avg Loss $",       "avg_loss",      False, False),
            ("Total Trades",     "total_trades",  None,  False),
        ]

        for i, (label, key, hb, is_pct) in enumerate(metrics_to_show):
            ov = old_result.get(key, "--")
            nv = new_result.get(key, "--")
            if isinstance(ov, float): ov = round(ov, 2)
            if isinstance(nv, float): nv = round(nv, 2)
            if hb is not None:
                metric_row(i+2, label, ov, nv, higher_better=hb, is_pct=is_pct)
            else:
                # Neutral metric (trades)
                tk.Label(body, text=label, font=F_LBL,
                         bg=S["bg"], fg=S["subtext"],
                         width=20, anchor="w").grid(row=i+2, column=0, padx=8, pady=3)
                tk.Label(body, text=str(ov), font=F_VAL,
                         bg=S["bg"], fg=S["orange"],
                         width=14, anchor="center").grid(row=i+2, column=1, padx=8)
                tk.Label(body, text=str(nv), font=F_VAL,
                         bg=S["bg"], fg=S["text"],
                         width=22, anchor="center").grid(row=i+2, column=2, padx=8)

        # ── Changed settings summary ───────────────────────────────
        changed_keys = [k for k in new_cfg
                        if str(new_cfg.get(k)) != str(old_cfg.get(k))
                        and k not in ("lookback_period","refresh_seconds",
                                      "discord_webhook","default_bt_days")]
        if changed_keys:
            tk.Frame(popup, bg=S["border"], height=1).pack(
                fill="x", padx=20, pady=(8,4))
            tk.Label(popup, text="CHANGED SETTINGS",
                     font=F_SEC, bg=S["bg"], fg=S["subtext"]).pack(anchor="w", padx=20)
            changes_txt = "  ".join(
                [f"{k}: {old_cfg.get(k)} → {new_cfg.get(k)}"
                 for k in changed_keys[:6]])
            tk.Label(popup, text=changes_txt,
                     font=("Courier",7), bg=S["bg"], fg=S["purple"],
                     wraplength=640, justify="left").pack(anchor="w", padx=20, pady=(2,6))

        # ── Buttons ────────────────────────────────────────────────
        tk.Frame(popup, bg=S["border"], height=1).pack(fill="x", padx=20, pady=(4,0))
        btn_row = tk.Frame(popup, bg=S["bg"])
        btn_row.pack(pady=14)

        def accept():
            popup.destroy()

        def undo():
            # Restore old config and strategy settings
            CONFIG.update(self._prev_config)
            STRATEGIES[ACTIVE_STRATEGY]["settings"].update(
                self._prev_strategy_settings)
            # Refresh settings form fields
            if hasattr(self, "_setting_vars"):
                for key, (var, kind) in self._setting_vars.items():
                    val = self._prev_config.get(key, "")
                    if val != "": var.set(str(val))
            # Restore old backtest display
            if self._prev_bt_result:
                old = self._prev_bt_result
                self._draw_backtest_chart(
                    old["df"], old["equity_curve"],
                    old.get("stop_curve", []), old["trades_raw"], old)
                self._update_stats_panel(old)
                self._populate_trade_list(old["trades_raw"])
                self.bt_result = old
            popup.destroy()
            if hasattr(self, "lbl_settings_status"):
                self.lbl_settings_status.config(
                    text="Changes undone — previous settings restored",
                    fg=S["orange"])

        tk.Button(btn_row, text="  ✓  Keep New Settings  ",
                  font=("Arial",11,"bold"),
                  bg=S["green"], fg=S["bg"],
                  relief="flat", cursor="hand2",
                  command=accept).pack(side="left", padx=12)

        tk.Button(btn_row, text="  ✕  Undo Changes  ",
                  font=("Arial",11,"bold"),
                  bg=S["red"], fg=S["bg"],
                  relief="flat", cursor="hand2",
                  command=undo).pack(side="left", padx=12)

    def _reset_settings(self):
        """Reset form fields to the active strategy's stored defaults."""
        strat_cfg = STRATEGIES[ACTIVE_STRATEGY]["settings"]
        for key, (var, kind) in self._setting_vars.items():
            # Use strategy default, fall back to CONFIG, fall back to empty
            val = strat_cfg.get(key, CONFIG.get(key, ""))
            if val != "":
                var.set(str(val))
        self.lbl_settings_status.config(
            text=f"Fields reset to '{ACTIVE_STRATEGY}' defaults", fg=S["subtext"])

    def _rerun_current_backtest(self):
        """Re-execute backtest on the last selected date range."""
        if self.bt_result is not None:
            self._execute_backtest(self.bt_result["start"], self.bt_result["end"])
        elif self.df is not None:
            self._run_default_backtest()

    def _on_strategy_change(self, name):
        """Switch active strategy — load its settings into CONFIG and rerun."""
        global ACTIVE_STRATEGY
        ACTIVE_STRATEGY = name

        # Load this strategy's settings into CONFIG
        strat_cfg = STRATEGIES[name]["settings"]
        CONFIG.update(strat_cfg)

        # If switching to Futures, auto-load the futures symbol
        if name == "Futures":
            futures_sym = strat_cfg.get("futures_symbol", "MES=F")
            CONFIG["ticker"] = futures_sym
            if hasattr(self, "lbl_ticker"):
                self.lbl_ticker.config(text=f" {futures_sym} ")
            self.title(f"Rush Algo's  —  {futures_sym}")

        # Rebuild settings tab
        if hasattr(self, "tab_settings"):
            for w in self.tab_settings.winfo_children():
                w.destroy()
            self._build_settings_tab(self.tab_settings)
        if hasattr(self, "lbl_settings_strat"):
            self.lbl_settings_strat.config(text=f"  —  {name}")
        if hasattr(self, "lbl_settings_status"):
            self.lbl_settings_status.config(
                text=f"Switched to '{name}' — settings loaded", fg=S["purple"])

        # Check if ticker or interval changed
        interval_changed = strat_cfg.get("interval") != CONFIG.get("interval")
        ticker_changed   = name == "Futures"  # always reload data for futures

        if interval_changed or ticker_changed:
            CONFIG["lookback_period"] = max_period_for_interval(CONFIG["interval"])
            self.df = None
            self._pending_rerun = True
            self.fetcher.stop()
            self.fetcher = DataFetcher(self.data_queue, CONFIG)
            self.fetcher.start()
            self.lbl_statusbar.config(
                text=f"Switched to {name} — fetching {CONFIG['ticker']} {CONFIG['interval']} data...")
        else:
            self._rerun_current_backtest()
            self.lbl_statusbar.config(
                text=f"Strategy: {name}  |  {CONFIG['ticker']}")

    def _ticker_prompt(self):
        popup = tk.Toplevel(self)
        popup.title("Change Ticker")
        popup.configure(bg=S["bg"])
        popup.resizable(False, False)
        popup.geometry(f"320x170+{self.winfo_x()+540}+{self.winfo_y()+60}")
        popup.grab_set()

        tk.Label(popup, text="Enter Ticker Symbol",
                 font=("Courier",11,"bold"), bg=S["bg"], fg=S["blue"]
                 ).pack(pady=(18,4))
        tk.Label(popup, text="e.g.  SPY  TSLA  BTC-USD  NVDA",
                 font=("Courier",8), bg=S["bg"], fg=S["subtext"]).pack()

        var = tk.StringVar(value=CONFIG["ticker"])
        ent = tk.Entry(popup, textvariable=var, font=("Courier",14,"bold"),
                       bg=S["panel"], fg=S["yellow"],
                       insertbackground=S["yellow"],
                       relief="flat", justify="center", width=12)
        ent.pack(pady=8)
        ent.select_range(0, "end")
        ent.focus_set()

        err = tk.Label(popup, text="", font=("Courier",8), bg=S["bg"], fg=S["red"])
        err.pack()

        def confirm(e=None):
            t = var.get().strip().upper()
            if not t: err.config(text="Enter a ticker"); return
            CONFIG["ticker"] = t
            self.lbl_ticker.config(text=f" {t} ")
            self.title(f"Rush Algo's  —  {t}")
            self._session_start  = None
            self._pending_rerun  = True
            self.df              = None   # clear stale df so calendar doesn't crash
            self.fetcher.stop()
            self.fetcher = DataFetcher(self.data_queue, CONFIG)
            self.fetcher.start()
            self.lbl_statusbar.config(text=f"Loading {t}...")
            popup.destroy()

        ent.bind("<Return>", confirm)
        bf = tk.Frame(popup, bg=S["bg"])
        bf.pack(pady=6)
        tk.Button(bf, text="Load", font=MONO_B,
                  bg=S["blue"], fg=S["bg"],
                  relief="flat", cursor="hand2",
                  command=confirm).pack(side="left", padx=6)
        tk.Button(bf, text="Cancel", font=("Courier",9),
                  bg=S["panel"], fg=S["subtext"],
                  relief="flat", cursor="hand2",
                  command=popup.destroy).pack(side="left", padx=6)
